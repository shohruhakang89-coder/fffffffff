def stretch(pw, salt):
    k = f"{salt}:{SECRET}".encode()
    st = pyhmac.new(k, pw.encode(), hashlib.sha256).digest()
    for r in range(4096):
        st = pyhmac.new(k, st + r.to_bytes(4,"big"), hashlib.sha256).digest()
    return st.hex()

def sign_token(uid, uname, exp):
    pay = f"{uid}.{uname}.{exp}"
    sig = pyhmac.new(SECRET.encode(), pay.encode(), hashlib.sha256).hexdigest()
    return f"{pay}.{sig}"

def verify_token(tok):
    p = tok.split(".")
    if len(p) != 4: return None
    pay = ".".join(p[:3])
    exp = pyhmac.new(SECRET.encode(), pay.encode(), hashlib.sha256).hexdigest()
    if not pyhmac.compare_digest(exp, p[3]): return None
    try:
        if int(p[2]) < time.time(): return None
        return p[0]
    except: return None

def db(sql, p=None, fetch=False):
    c = pool_.getconn()
    try:
        with c:
            with c.cursor(cursor_factory=RealDictCursor) as cur:
                if p is not None: cur.execute(sql, p)
                else: cur.execute(sql)
                if fetch: return [dict(r) for r in cur.fetchall()]
    finally: pool_.putconn(c)

def hav(lon1, lat1, lon2, lat2):
    R=6371000; p1=math.radians(lat1); p2=math.radians(lat2)
    a=math.sin((p2-p1)/2)**2+math.cos(p1)*math.cos(p2)*math.sin(math.radians(lon2-lon1)/2)**2
    return 2*R*math.asin(math.sqrt(min(1.0,a)))

def trail_dist(trail):
    return sum(hav(float(trail[i-1][0]),float(trail[i-1][1]),float(trail[i][0]),float(trail[i][1])) for i in range(1,len(trail)) if len(trail[i])==2)

                cam_lon=clon if follow else cx; cam_lat=clat if follow else cy

@app.route("/api/chat/list")
def chatlist():
    u = get_user()
    if not u: return jsonify({"ok":False}), 401
    rows = db("""SELECT DISTINCT ON (other_id) other_id, username, color, text, photo_path, video_path, created_at
        FROM (SELECT CASE WHEN from_id=%s THEN to_id ELSE from_id END AS other_id, u.username, u.color, m.text, m.photo_path, m.video_path, m.created_at
              FROM chat_messages m JOIN users u ON u.id=(CASE WHEN m.from_id=%s THEN m.to_id ELSE m.from_id END)
              WHERE m.from_id=%s OR m.to_id=%s) t ORDER BY other_id, created_at DESC""",(u["id"],u["id"],u["id"],u["id"]),fetch=True)
    unread = db("SELECT from_id, COUNT(*) AS c FROM chat_messages WHERE to_id=%s AND read_at=0 GROUP BY from_id",(u["id"],),fetch=True)
    umap = {r["from_id"]:r["c"] for r in unread}
    conv = [dict(r) for r in rows]
    for c in conv: c["unread"] = umap.get(c["other_id"], 0)
    return jsonify({"ok":True,"conversations":conv})
