def norm(w: str) -> str:
    """So'zni taqqoslash uchun normalizatsiya qiladi: barcha apostrof
    belgilari O'CHIRILADI va kichik harfga o'tkaziladi.

    v7.5 — MUHIM: STT/diktofon matnlarida apostroflar ko'pincha YO'Q
    ('bolgan', 'tortinchidan', 'soradi'), yozma matnda esa bor
    ('bo'lgan', 'to'rtinchidan', 'so'radi'). Lug'atlarda taqqoslash
    apostrofsiz kalitlar orqali boradi — ikkala yozuv ham bir xil taniladi.
    """
    return _APOS_RE.sub("", w).lower()

def _keyed(s):
    """To'plam/dikt/tuple elementlarini apostrofsiz kalitlarga o'tkazadi."""
    if isinstance(s, dict):
        return {norm(k): v for k, v in s.items()}
    if isinstance(s, tuple):
        return tuple(norm(x) for x in s)
    if isinstance(s, (set, frozenset)):
        if s and all(isinstance(x, tuple) for x in s):
            return {tuple(norm(a) for a in x) for x in s}
        return {norm(x) for x in s}
    return s

def is_finite(w: str) -> bool:
    """So'z kesim (finite fe'l) shaklida ekanini taxmin qiladi."""
    # v7.2: ko'plik otlar 'vazifalar/qoidalar/masalalar' FIN_STRONG 'alar'
    # tufayli fe'l deb noto'g'ri tanilar edi - birinchi navbatda chiqaramiz
    if (w.endswith("lar") and not w.endswith(("alarqasi", "alaridagi"))
            and not w.endswith(("aladi", "alalar", "aladilar"))):
        return False
    # v7.2: '-imiz' egalik ('maqsadimiz', 'tashqarimiz') - fe'l emas
    if w.endswith("imiz") and len(w) >= 6:
        return False
    # v7.4: '-yati' egalikli arabizmlar ('ahamiyati', 'imkoniyati',
    # 'iqtisodiyati', 'qobiliyati') - ot, fe'l emas
    if w.endswith("yati"):
        return False
    # v7.2: '-lik' ot 'qulaylik/barqarorlik' FIN_STRONG 'aylik' tufayli
    # fe'l deb noto'g'ri tanilmasin (aslida 'qulay + lik' ot)
    if w.endswith("lik") and (w[:-3] in ADJ_WORDS or w[:-3] in PRED_ADJ):
        return False
    if w in AY_EXCEPTIONS:
        # 'qulay', 'shunday' kabi '-ay' bilan tugagan so'zlar fe'l EMAS
        # (v6 dagi 'qulay. Interfeys' soxta chegarasini tuzatadi)
        return False
    if w in FIN_EXCEPTIONS or w in QWORDS:
        # 'endi' ravishi, 'qanday/qalaysiz' so'roq so'zlari — fe'l emas
        return False
    # v7.5: kopula shakllari (shaxs qo'shimchali) ham kesim:
    # 'uxlayotgan edim', 'shart edim' — qo'shma kesim
    if w in {"edi", "edim", "edik", "eding", "edingiz", "edilar", "ekan",
             "emas", "bol", "yoq", "kerak"}:
        return True
    # '-miz' kesimi undoshdan keyin: 'xursandmiz', 'tayyormiz' -
    # o'zgalik '-imiz'/-'amiz' unlidan keyin: 'loyihamiz', 'kitobimiz'
    if w.endswith("miz") and len(w) >= 5 and w[-4] not in "aeiouo\u00f6\u00fc":
        return True
    # v7.4: predikativ sifat + '-man' kesimi: 'mamnunman', 'xursandman',
    # 'tayyorman', 'bandman' (undoshdan keyin '-man')
    if w.endswith("man") and len(w) >= 6 and w[-4] not in "aeiouo\u00f6\u00fc":
        return True
    # niyat fe'li + kesim: 'olmoqchiman', 'kelmoqchimiz', 'qurmoqchidilar'
    if ("moqchi" in w and
            w.endswith(("man", "miz", "siz", "dilar", "manlar", "mizlar",
                        "sizlar", "sin", "sinlar"))):
        return True
    for suf in FIN_STRONG:
        if w.endswith(suf) and len(w) - len(suf) >= 2:
            return True
    return False


def person_of(w: str) -> int:
    """Kesim shaklidan shaxsni aniqlaydi: 1, 2, 3; aniqlanmasa 0."""
    for suf in ("ganiman", "diman", "aman", "yapman", "man", "ganmiz", "dimiz",
                "amiz", "yapmiz", "miz"):
        if w.endswith(suf):
            return 1
    for suf in ("gansiz", "dingiz", "asiz", "yapsiz", "siz", "sizlar", "ngiz",
                "nglar", "ginsiz", "sangiz", "ding", "yapsan", "gansan", "asan", "san"):
        if w.endswith(suf):
            return 2
    for suf in ("dilar", "ganlar", "apti", "yapti", "moqda", "moqchi", "ekan",
                "edi", "adi", "ydi", "gan", "di"):
        if w.endswith(suf):
            return 3
    return 0


def has_cond_suffix(w: str) -> bool:
    if w in COND_BLOCK or w in QWORDS:
        return False
    for suf in COND_SUFFIXES:
        if w.endswith(suf) and len(w) - len(suf) >= 3:
            return True
    return False


def has_converb_suffix(w: str) -> bool:
    if w in ("g'olib",):  # ot: g'olib (converb emas)
        return False
    for suf in CONVERB_SUFFIXES:
        if w.endswith(suf):
            stem_len = len(w) - len(suf)
            if stem_len >= (2 if suf == "ib" else 3):  # "olib", "qilib"...
                return True
    return False


def is_vchain_aux(w: str) -> bool:
    """So'z fe'l birikmasining yordamchi fe'lmi: 'boshladi', 'yubordi',
    'qo'yishimiz', 'bering', 'chiqdim'... (S6 qo'riqlanadi)."""
    for stem in VCHAIN_STEMS:
        if w.startswith(stem) and len(w) >= len(stem) + 2:
            return True
    return False


def is_converb_like(w: str) -> bool:
    """So'z fe'l birikmasining BIRINCHI elementi bo'la oladimi:
    '-ib', '-y'/'-ay' (hozirgi zamon ravishdoshi), '-a'..."""
    return w.endswith(("ib", "ay", "y", "a")) and len(w) >= 3


def masdar_like(w: str) -> bool:
    """So'z masdar (-ish/-moq) yoki uning qo'shimchalimi: 'yaratish',
    'aniqlashimiz'... (gap oxiri/boshi bo'la oladi)."""
    return (w.endswith(("ish", "ishni", "ishga", "ishda", "ishimiz",
                        "ishingiz", "ishlari", "moq", "moqda"))
            and len(w) >= 5)


def is_complement(w: str) -> bool:
    """Qaram gap (ergash gap) kesimini o'z ichiga olgan shakl: 'kelganini'."""
    for suf in COMPLEMENT_ENDINGS:
        if w.endswith(suf) and len(w) - len(suf) >= 3:
            return True
    return False


def end_class(w: str) -> str:
    """Uyushiq bo'laklarning qo'shimcha sinfini topadi (o'xshashlik uchun)."""
    for suf in ("larini", "lariga", "larida", "laridan", "laringiz", "larimiz",
                "larning", "larim", "lari", "lar", "ning", "dan", "ga", "da",
                "ni", "im", "ing", "si", "i", "miz", "ngiz", "imiz", "ingiz"):
        if w.endswith(suf):
            return suf
    return ""

@dataclass
class Edit:
    """Matnga taklif qilingan bitta tahrir."""
    start: int
    end: int
    text: str
    rule: str = ""
    cat: str = ""
    name: str = ""
    msg: str = ""
    conf: float = 0.0
    base: str = ""  # hisobot uchun: tahlil qilingan matn (snippet ko'rsatishda)

    def __post_init__(self):
        # Himoya: positional chaqiruvda conf msg o'rniga tushib qolsa — tuzatamiz
        if isinstance(self.msg, (int, float)) and not isinstance(self.msg, bool):
            self.conf = float(self.msg)
            self.msg = str(self.name or "")
            self.name = self.cat


@dataclass
class Tok:
    """Token: so'z yoki tinish belgisi (absolyut offsetlar bilan)."""
    text: str
    start: int
    end: int
    lower: str = ""

    def __post_init__(self):
        if not self.lower:
            self.lower = norm(self.text)

    @property
    def is_word(self) -> bool:
        return bool(re.match(r"[^\W\d_]", self.text))  # harf bilan boshlanadi

    @property
    def is_num(self) -> bool:
        return bool(re.match(r"\d", self.text))


@dataclass
class Sentence:
    """Gap: matn, chegaralar va tahlil natijalari."""
    text: str
    start: int
    end: int
    analysis: Dict[str, Any] = field(default_factory=dict)

    @property
    def stripped(self) -> str:
        return self.text.rstrip()

    @property
    def terminator(self) -> str:
        t = self.stripped
        return t[-1] if (t and t[-1] in ".!?…,") else ""


@dataclass
class Result:
    """To'liq natija: tuzatilgan matn + izohlar + ogohlantirishlar."""
    original: str
    corrected: str
    findings: List[Edit]      # qo'llanilgan o'zgarishlar
    warnings: List[Edit]      # shubhali/tavsiya etilgan, LEKIN qo'llanilmagan
    stats: Dict[str, Any] = field(default_factory=dict)

    def as_dict(self) -> Dict[str, Any]:
        def clean(e: Edit) -> Dict[str, Any]:
            d = asdict(e)
            d.pop("base", None)
            return d
        return {
            "original": self.original,
            "corrected": self.corrected,
            "findings": [clean(f) for f in self.findings],
            "warnings": [clean(w) for w in self.warnings],
            "stats": self.stats,
        }

# 5. HIMOYALANGAN ZONALAR (ProtectedSpans)
# ==============================================================================

_MONTHS = (r"(?:yanvar|fevral|mart|aprel|may|iyun|iyul|avgust|sentabr|sentyabr"
           r"|oktabr|oktyabr|noyabr|dekabr)")

PROTECTED_RES = [
    re.compile(r"https?://\S+|www\.\S+"),
    re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+"),
    re.compile(r"[#@]\w+"),
    re.compile(r"\+?\d[\d\s()-]{7,}\d"),                                   # telefon
    re.compile(r"\b\d{1,2}[:.]\d{2}\b"),                                   # vaqt 14:30
    re.compile(r"\b\d+(?:[.,]\d+)+\b"),                                    # kasr 3.14
    re.compile(r"\b\d{1,3}(?:,\d{3})+\b"),                                 # 1,000,000
    re.compile(r"\b\d{4}(?:-\d{2}){0,2}\b"),                               # 2025-11-15
    re.compile(r"\b\d{1,2}-" + _MONTHS + r"\b", re.I),                     # 15-may
    re.compile(r"\b\d+-yil(?:larda|lardagi|da|dagi|ning)?\b", re.I),       # 2025-yil
    re.compile(r"\b\d+-(?:kun|sonli|kurs|sinf|bosqich|raqam|qavat|bet"
               r"|tomonlama|minut|soat)\b", re.I),
    re.compile(r"\b(?:v\.h\.k?\.?|o['\u02bb\u2019]?r\.|t\.kr\.|q\.b\.|tb\."
               r"|asr\.|prof\.|akad\.|dots\.|tahr\.|nashr\.|mil\.|av\.|sm\.|km\.|kg\.)",
               re.I),
    re.compile(r"\b(?:O['\u02bb\u2019]?|[A-Z])\.(?=\s+[A-Z\u02bb\u2019'O])"),  # A. Qodiriy
    re.compile(r"(?m)^\s*\d+\.\s"),                                        # ro'yxat "1. "
]


def find_protected(text: str) -> List[Tuple[int, int]]:
    spans = []
    for rx in PROTECTED_RES:
        for m in rx.finditer(text):
            spans.append((m.start(), m.end()))
    return spans


def in_protected(pos: int, spans: List[Tuple[int, int]]) -> bool:
    return any(s <= pos < e for s, e in spans)


def overlaps_protected(e: Edit, spans: List[Tuple[int, int]]) -> bool:
    return any(not (e.end <= s or e.start >= pe) for s, pe in spans)


# ==============================================================================
# 6. TOKENIZER VA GAP AJRATUVCHI
# ==============================================================================

TOKEN_RE = re.compile(r"[\w'-]+|[^\s\w]", re.UNICODE)
_BOUNDARY_RE = re.compile(r"[.!?…]+[)\]»\"']*", re.UNICODE)


def tokenize(text: str) -> List[Tok]:
    return [Tok(m.group(0), m.start(), m.end()) for m in TOKEN_RE.finditer(text)]


def split_sentences(text: str, protected: List[Tuple[int, int]]) -> List[Sentence]:
    """Matnni gaplarga bo'ladi: qisqartma, kasr, sana va ro'yxatlarni himoya qiladi."""
    sentences: List[Sentence] = []
    offset = 0
    for line in text.split("\n"):
        sentences.extend(_split_line(line, offset, protected))
        offset += len(line) + 1
    return sentences


def _split_line(line: str, offset: int, protected: List[Tuple[int, int]]) -> List[Sentence]:
    out: List[Sentence] = []
    n = len(line)
    if not line.strip():
        return out
    start = 0
    for m in _BOUNDARY_RE.finditer(line):
        end = m.end()
        rest = line[end:]
        stripped = rest.lstrip()
        nxt = stripped[:1]
        valid = False
        if stripped == "":
            valid = True
        elif nxt and (nxt.isalpha() or nxt in "\"'«([–—-" or nxt.isdigit()):
            valid = True
        # Nuqta himoyalangan zonada (qisqartma/kasr/sana/ro'yxat) — chegara EMAS
        if valid and in_protected(offset + m.start(), protected):
            valid = False
        if valid and nxt.isdigit():
            valid = False  # "3.14" yoki "12. 13." raqam ketma-ketligi
        if valid:
            out.append(Sentence(line[start:end], offset + start, offset + end))
            start = end + (len(rest) - len(stripped))
    if start < n and line[start:].strip():
        out.append(Sentence(line[start:], offset + start, offset + len(line)))
    return out


# ==============================================================================
# 7. DVIGATEL: SO'ROQ BELGISI TAHLILI
# ==============================================================================

WARN_FLOOR = 0.45          # bundan past takliflar umuman ko'rsatilmaydi
SAFE_THRESHOLD = 0.70      # safe rejimda qo'llaniladigan minimal ishonch
AGGR_THRESHOLD = 0.45      # aggressive rejim chegarasi

RULE_PRIORITY = {  # bir xil ishonchdagi to'qnashuvda ustuvorlik (kichigi yutadi)
    "T2": 0, "V15": 1, "T3": 2, "T1": 3, "T4": 4,
}


class UzbekGrammar:
    """Training-free o'zbek punktuatsiya dvigateli."""

    def __init__(self, mode: str = "safe", apostrophe: str = "keep"):
        if mode not in ("safe", "aggressive"):
            raise ValueError("mode: 'safe' yoki 'aggressive'")
        self.mode = mode
        self.apostrophe = apostrophe
        self.threshold = SAFE_THRESHOLD if mode == "safe" else AGGR_THRESHOLD
        self._text = ""

    # ------------------------------------------------------------- yordamchilar

    def _cb(self, tok: Tok) -> str:
        """Tokendan oldingi belgi."""
        i = tok.start - 1
        return self._text[i] if i >= 0 else ""

    def _ca(self, tok: Tok) -> str:
        """Tokendan keyingi belgi."""
        i = tok.end
        return self._text[i] if i < len(self._text) else ""

    def _words(self, toks: List[Tok]) -> List[Tok]:
        return [t for t in toks if t.is_word]

    def _last_word(self, toks: List[Tok]) -> Optional[Tok]:
        for t in reversed(toks):
            if t.is_word:
                return t
        return None

    def _comma_before(self, tok: Tok, conf: float, rule: str, name: str,
                      msg: str) -> Optional[Edit]:
        # Vergul token oralig'idagi bo'shliqlardan OLDINGI so'zga yopishtiriladi
        i = tok.start
        while i > 0 and self._text[i - 1] in " \t":
            i -= 1
        if i > 0 and self._text[i - 1] == ",":
            return None
        # Gap oxiri belgisidan keyin vergul qo'ymaymiz: "kerak. Shuning"
        if i > 0 and self._text[i - 1] in ".!?…:;":
            return None
        return Edit(i, i, ",", rule, "Vergul", name, msg, conf)

    def _comma_after(self, tok: Tok, conf: float, rule: str, name: str, msg: str,
                     toks: Optional[List[Tok]] = None) -> Optional[Edit]:
        # Keyingi bo'sh bo'lmagan belgi vergul bo'lsa — allaqachon bor
        j = tok.end
        while j < len(self._text) and self._text[j] in " \t":
            j += 1
        if j < len(self._text) and self._text[j] == ",":
            return None
        # Keyingi belgi gap oxiri bo'lsa vergul qo'ymaymiz: "do'stim!" / "keldi."
        if j < len(self._text) and self._text[j] in ".!?…:;":
            return None
        # Keyingi so'z vergul oladigan bog'lovchi bo'lsa — qo'sh vergul oldini olamiz
        if toks is not None:
            for t in toks:
                if t.start >= tok.end:
                    if t.is_word and t.lower in COMMA_TAKERS:
                        return None
                    break
        return Edit(tok.end, tok.end, ",", rule, "Vergul", name, msg, conf)

    def _nounish(self, t: Tok) -> bool:
        w = t.lower
        return (
            t.is_word and not t.is_num
            and not is_finite(w)
            and w not in FUNCTION_WORDS
            and w not in PLACE_SUFFIXES
            and w not in QUYIDA_SET
            and w not in KIRISH_STRONG
            and w not in KIRISH_WEAK
            and w not in ORDINAL_KIRISH
            and w not in INTERJECTIONS
            and w not in SALUTATIONS
            and w not in TIME_ADVERBS
            and not w.endswith("dagi")
            and not has_converb_suffix(w)
        )

    def _adjlike(self, t: Tok) -> bool:
        w = t.lower
        if w in ADJ_WORDS:
            return True
        return any(w.endswith(s) for s in
                   ("li", "lik", "liq", "siz", "gi", "gina", "roq", "iy"))

    # ------------------------------------------------- v7.5: qattiq himoyalar

    def _reported_speech(self, words: List[Tok]) -> bool:
        """Gap o'zlashtirma nutqmi? ('nega kechikding deb soradi',
        '... deb aytdi', '... degan jumla') — 'deb/degan' birikmasidan
        keyin 3 so'z ichida kesim/nutq fe'li bo'lsa — ha."""
        n = len(words)
        for i in range(n):
            w = words[i].lower
            if w in ("deb", "degan", "degani", "deganda", "dediki", "derdi",
                     "deganingiz", "deganimda"):
                for u in range(i + 1, min(n, i + 4)):
                    w2 = words[u].lower
                    if (is_finite(w2) or w2 in SPEECH_VERBS
                            or w2 in ("aytmoq", "soramoq")):
                        return True
                return False
        return False

    def _never_start_word(self, w: str) -> bool:
        """So'z yangi mustaqil gapni boshlay oladimi? v7.5: qattiq lingvistik
        himoya — quyidagi so'zlardan OLDIN gap-chegarasi QO'YILMAYDI:
        kopula/yordamchi kesimlar, o'zlashtirma nutq birliklari ('deb/degan'),
        qaram bog'lovchilar ('va', 'ammo', 'chunki'), qaram kesim shakllari
        ('ekanini'), ravishdoshlar ('gaplashib') va shart shakllari ('bolsa')."""
        k = norm(w)
        if k in NEVER_START:
            return True
        if k in POSTPOSITIONS:
            return True
        if is_complement(k):
            return True
        if k in SUBJ_AUX_CONT:
            return True
        # ravishdosh shakl — mustaqil gap boshi bo'lmaydi ('gaplashib ...')
        if has_converb_suffix(k) and not self._is_imperative_word(k):
            return True
        # shart shakli ('bolsak', 'ajratsa') — ergash gap boshi
        if has_cond_suffix(k):
            return True
        return False

    # ------------------------------------------------- so'roq belgisi tahlili

    def _question_signal(self, s: Sentence, words: List[Tok]):
        """Gap so'roq gapmi? (True/False/None=aniq emas) + sabab + ishonch."""
        if not words:
            return None, "", 0.0
        cached = s.analysis.get("q")
        if cached is not None:
            return cached
        fw = words[0].lower
        lw = words[-1].lower
        bigram_q = fw == "nima" and len(words) > 1 and words[1].lower == "uchun"
        any_q = bigram_q or any(is_qword_at(words, i)
                                for i in range(len(words)))
        result = (None, "", 0.0)
        if any_q:
            # v7.2: oxirgi so'zdagi '-mi' yuklamasi KUCHLI dalil - qaram gap
            # gardidan KUCHLIROQ: 'server qanday ishlashini tekshirib
            # ko'rdingizmi?' - ichida qaram so'roq bo'lsa ham bu so'roq gap.
            if lw.endswith("mi") and len(lw) > 2:
                stem = lw[:-2]
                if (is_finite(stem) or stem in MI_STEMS
                        or stem in TIME_ADVERBS or stem in PRED_NOMINAL):
                    result = (True, "oxirgi so'zda '-mi' so'roq yuklamasi", 0.88)
                    s.analysis["q"] = result
                    return result
            # Qaram gap: "Bilaman kim kelganini." — so'roq belgisi kerak emas
            if (fw in KNOW_VERBS or lw in KNOW_VERBS
                    or any(is_complement(w.lower) for w in words)):
                result = (None, "So'roq so'zi qaram (ergash) gap tarkibida — "
                                "so'roq belgisi talab etilmaydi", 0.0)
                s.analysis["q_info"] = True
            elif (fw not in QWORDS and not bigram_q
                    and any(w.lower in KNOW_VERBS for w in words)):
                # "kim biladi degan savol tug'iladi" — o'rindosh so'roq
                ki = next(i for i, w in enumerate(words)
                          if w.lower in KNOW_VERBS)
                if len(words) - ki - 1 >= 3:
                    result = (None, "So'roq so'zi o'rindosh gap tarkibida "
                                    "('kim biladi degan...') — so'roq belgisi "
                                    "talab etilmaydi", 0.0)
                    s.analysis["q_info"] = True
            elif fw in QWORDS or bigram_q:
                # "Qanday chiroyli!" — undov ma'nosi ham mumkin
                if (fw == "qanday" and len(words) > 1 and self._adjlike(words[1])
                        and not any(is_finite(w.lower) for w in words[1:])):
                    result = (False, "'qanday' undov ma'nosida kelgan", 0.0)
                elif self._reported_speech(words):
                    # v7.5: o'zlashtirma nutq: 'nega kechikding deb soradi' —
                    # so'roq belgisi QO'YILMAYDI
                    result = (None, "So'roq so'zi o'zlashtirma nutq ('deb/degan') "
                                    "tarkibida — so'roq belgisi kerak emas", 0.0)
                else:
                    result = (True, "gap boshida so'roq so'zi ('%s')" % words[0].text, 0.92)
            else:
                # v7.5: so'roq so'zi gap o'rtasida — ko'pincha bayon gap.
                # So'roq bo'ladi AGAR: so'roq so'zi gap boshiga yaqin,
                # undan keyin qaram-gap kesimi ('... ishlashini') yo'q,
                # 'qanday ... kabi' o'xshatish yo'q va o'zlashtirma yo'q:
                #   ✓ 'Loyihani qancha vaqtda tiklay olamiz?'
                #   ✗ '... qanday munosabatda ishlashini ham tekshirish zarur.'
                #   ✗ '... kim bilandir gaplashib ketayotgan edi.'
                near_q, sub_q = False, False
                for t in range(min(7, len(words))):
                    if is_qword_at(words, t):
                        near_q = True
                        tail = [words[u].lower for u in range(t + 1, len(words))]
                        if any(is_complement(x) for x in tail):
                            sub_q = True
                        if any(x in ("kabi", "singari") for x in tail[:8]):
                            sub_q = True
                        break
                if sub_q or self._reported_speech(words):
                    result = (None, "So'roq so'zi qaram gap/o'zlashtirma "
                                    "tarkibida — so'roq belgisi kerak emas", 0.0)
                elif near_q and (lw in QWORDS
                                 or any(is_finite(w.lower) for w in words[1:])):
                    result = (True, "gap boshiga yaqin so'roq so'zi ('%s') "
                                    "bilan so'roq gap" % words[-1].text, 0.8)
                else:
                    result = (None, "So'roq so'zi gap o'rtasida — bayon gap, "
                                    "so'roq belgisi talab etilmaydi", 0.0)
        elif lw == "mi":
            result = (True, "alohida '-mi' so'roq yuklamasi", 0.88)
        elif lw.endswith("mi") and len(lw) > 2:
            stem = lw[:-2]
            if (is_finite(stem) or stem in MI_STEMS or stem in TIME_ADVERBS
                    or stem in PRED_NOMINAL):
                result = (True, "oxirgi so'zda '-mi' so'roq yuklamasi", 0.88)
        s.analysis["q"] = result
        return result

    # ======================================================================
    # 8. ASOSIY METOD: correct()
    # ======================================================================

    def correct(self, text: str) -> Result:
        if not text or not text.strip():
            return Result(text or "", text or "", [], [],
                          {"changes": 0, "warnings": 0, "sentences": 0,
                           "mode": self.mode, "version": __version__})
        text = text.replace("\r\n", "\n").replace("\r", "\n")
        findings: List[Edit] = []
        warnings: List[Edit] = []

        # --- 1-BASQICH: global normallash (N-qoidalari)
        prot0 = find_protected(text)
        edits = [e for e in self._norm_edits(text, prot0)
                 if not overlaps_protected(e, prot0)]
        chosen, dropped = self._resolve(edits)
        for e in chosen + dropped:
            e.base = text
        text1 = self._apply(text, chosen)
        findings += chosen
        warnings += [e for e in dropped if e.conf >= WARN_FLOOR]

        # --- 1.4-BOSQICH: NUTQ (qo'shtirnoq) FAZASI (v7)
        # To'g'ridan-to'g'ri nutqni run-on segmentatsiyasidan oldin
        # tartibga solamiz: 'aytdi: "..."' ikki nuqtasi, nutq oxiriga
        # nuqta/so'roq belgisi (qo'shtirnoq ichiga), nutq boshi bosh harf.
        qprot = find_protected(text1)
        qedits = [e for e in self._quote_pass_edits(text1)
                  if not overlaps_protected(e, qprot)]
        if qedits:
            self._text = text1
            chosen_q, dropped_q = self._resolve(qedits)
            for e in chosen_q + dropped_q:
                e.base = text1
            text1 = self._apply(text1, chosen_q)
            findings += chosen_q
            warnings += [e for e in dropped_q if e.conf >= WARN_FLOOR]

        # --- 1.5-BASQICH: RUN-ON SEGMENTATSIYA
        # Tinish belgisi yo'q uzun gaplar (STT/diktofon matni) tahlil
        # qilinadi: ishonchli joylar nuqta/so'roq belgisi bilan gaplarga
        # bo'linadi (S1/S2/S3, T9, T10). Punksiyatsiyali matn O'ZGARMAYDI
        # va qayta ishlovda barqaror (idempotent) qoladi.
        text1b = text1
        self._text = text1
        prot_s = find_protected(text1)
        sents0 = split_sentences(text1, prot_s)
        toks0 = tokenize(text1)
        seg_edits: List[Edit] = []
        for s0 in sents0:
            stoks0 = [t for t in toks0 if s0.start <= t.start < s0.end]
            if stoks0:
                seg_edits.extend(self._segment_runon(s0, stoks0, prot_s))
        seg_edits = [e for e in seg_edits
                     if not overlaps_protected(e, prot_s)]
        if seg_edits:
            chosen_seg, dropped_seg = self._resolve(seg_edits)
            for e in chosen_seg + dropped_seg:
                e.base = text1
            text1b = self._apply(text1, chosen_seg)
            findings += chosen_seg
            warnings += [e for e in dropped_seg if e.conf >= WARN_FLOOR]

        # --- 2-BASQICH: gap bo'yicha qoidalar
        self._text = text1b
        prot1 = find_protected(text1b)
        sents = split_sentences(text1b, prot1)
        toks_all = tokenize(text1b)
        chosen_all: List[Edit] = []
        for s in sents:
            stoks = [t for t in toks_all if s.start <= t.start < s.end]
            if not stoks:
                continue
            sedits: List[Edit] = []
            for rid in self._rule_order():
                getattr(self, "_r_" + rid)(s, stoks, sedits)
            # v7: har bir gap ALOHIDA yechiladi (katta matnlarda tezlik)
            sedits = [e for e in sedits if not overlaps_protected(e, prot1)]
            chosen_s, dropped_s = self._resolve(sedits)
            for e in chosen_s + dropped_s:
                e.base = text1b
            chosen_all += chosen_s
            findings += chosen_s
            warnings += [e for e in dropped_s if e.conf >= WARN_FLOOR]
        text2 = self._apply(text1b, chosen_all)

        # --- 3-BASQICH: yakuniy tozalash (qoldiq ikkilangan belgilar)
        prot2 = find_protected(text2)
        edits = [e for e in self._final_edits(text2, prot2)
                 if not overlaps_protected(e, prot2)]
        chosen, dropped = self._resolve(edits)
        for e in chosen + dropped:
            e.base = text2
        text3 = self._apply(text2, chosen)
        findings += chosen
        warnings += [e for e in dropped if e.conf >= WARN_FLOOR]

        # --- Balans tekshiruvlari (faqat ogohlantirish)
        warnings += self._balance_checks(text3)

        return Result(
            original=text,
            corrected=text3,
            findings=findings,
            warnings=warnings,
            stats=self._stats(findings, warnings, len(sents)),
        )

    def _rule_order(self) -> List[str]:
        return [
            "T4", "T1", "T2", "T3", "T7", "T8",   # gap boshi/oxiri, nutq (v7), tire
            "V1", "V2", "V15", "V16",          # kirish so'z, bog'lanmagan so'z, undov
            "V3",                              # murojaat (undalma)
            "V4", "V5", "V6", "V7",            # bog'lovchilar
            "V8", "V9", "V12",                 # ergash gaplar
            "V10", "V11", "V18", "V19",          # sanash, ham..ham, qo'shma, appozitsiya (v7)
            "V13", "V17",                      # ergash sanash, '-ishi esa'
            "V21", "V22", "V23", "V24",      # v7.2: parall tuzilmalar sanashi
            "V14",                            # ya'ni
            "I1", "I2", "I3",                 # ikki nuqta
            "I4", "I5",                       # v7: sanash ikki nuqtasi, 'bor:', nuqta-vergul
            "AV1",                            # ortiqcha vergul
        ]

    # -------------------------------------------------- tahrirlarni qo'llash

    def _resolve(self, edits: List[Edit]):
        """Takliflarni saralaydi: to'qnashuvda eng ishonchlisi, keyin ustuvorlik.
        v7: katta matnlar (kitob hajmi) uchun to'qnashuv tekshiruvi logarifmik
        oynadan (bisect) foydalanadi - o'n minglab tahrirlarda ham tez."""
        order = sorted(
            edits,
            key=lambda e: (-e.conf, RULE_PRIORITY.get(e.rule, 50), e.start, e.end),
        )
        chosen: List[Edit] = []
        dropped: List[Edit] = []
        starts: List[int] = []          # chosen[].start bo'yicha saralangan
        max_span = 1                     # hozirgacha ko'rilgan eng uzun tahrir
        for e in order:
            clash = False
            if chosen:
                lo = bisect.bisect_left(starts, e.start - max_span)
                hi = bisect.bisect_right(starts, e.end)
                for idx in range(lo, hi):
                    c = chosen[idx]
                    # nuqta-tahrirlar (start == end) uchun <= muhim:
                    # bir xil pozitsiyaga ikkita vergul qo'yilmasin
                    if e.start <= c.end and c.start <= e.end:
                        clash = True
                        break
            if clash:
                e.msg = ("Boshqa qoida bilan to'qnashdi (o'zgartirilmadi). " + e.msg)
                dropped.append(e)
                continue
            # W-qoidalari (shubha/ma'lumot) hech qachon matnni o'zgartirmaydi
            if (e.conf >= self.threshold
                    and not (e.text == "" and e.rule.startswith("W"))):
                pos = bisect.bisect_right(starts, e.start)
                starts.insert(pos, e.start)
                chosen.insert(pos, e)
                max_span = max(max_span, e.end - e.start)
            else:
                dropped.append(e)
        chosen.sort(key=lambda e: (e.start, e.end))
        return chosen, dropped

    @staticmethod
    def _apply(text: str, edits: List[Edit]) -> str:
        out = []
        pos = 0
        for e in sorted(edits, key=lambda e: (e.start, e.end)):
            if e.start < pos:
                continue
            out.append(text[pos:e.start])
            out.append(e.text)
            pos = e.end
        out.append(text[pos:])
        return "".join(out)

    def _stats(self, findings: List[Edit], warnings: List[Edit], n_sents: int) -> Dict[str, Any]:
        by_cat: Dict[str, int] = {}
        for f in findings:
            by_cat[f.cat] = by_cat.get(f.cat, 0) + 1
        return {
            "version": __version__,
            "mode": self.mode,
            "threshold": self.threshold,
            "sentences": n_sents,
            "changes": len(findings),
            "warnings": len(warnings),
            "by_category": by_cat,
            "rules_applied": sorted({f.rule for f in findings}),
        }

    # ======================================================================
    # 9. N-QOIDALARI: GLOBAL NORMALLASH
    # ======================================================================

    def _norm_edits(self, text: str, prot: List[Tuple[int, int]]) -> List[Edit]:
        out: List[Edit] = []

        def add(rx, repl, rule, name, msg, conf):
            for m in rx.finditer(text):
                r = repl(m) if callable(repl) else repl
                out.append(Edit(m.start(), m.end(), r, rule, "Normallash",
                                name, msg, conf))

        # N1: ketma-ket bo'shliqlar
        add(re.compile(r"[ \t]{2,}"), " ", "N1", "Bo'shliqlar",
            "Ketma-ket bo'shliqlar bittaga keltirildi", 0.95)
        add(re.compile(r"[ \t]+\n"), "\n", "N1", "Bo'shliqlar",
            "Satr oxiridagi bo'shliq olib tashlandi", 0.9)
        # N2: tinish belgisi oldidagi bo'shliq
        add(re.compile(r"[ \t]+([,.;:!?…%)\]}»])"), lambda m: m.group(1), "N2",
            "Bo'shliq", "Tinish belgisi oldidagi ortiqcha bo'shliq olib tashlandi", 0.95)
        # N3: tinish belgilaridan keyin bitta bo'shliq
        add(re.compile(r"([,;:])(?=[^\s\d)])"), lambda m: m.group(1) + " ", "N3",
            "Bo'shliq", "Tinish belgisidan keyin bo'shliq qo'yildi", 0.9)
        add(re.compile(r"([!?…])(?=[A-Za-zА-Яа-я])"), lambda m: m.group(1) + " ", "N3",
            "Bo'shliq", "Tinish belgisidan keyin bo'shliq qo'yildi", 0.9)
        add(re.compile(r"\.(?=[A-Za-z])"), ". ", "N3", "Bo'shliq",
            "Nuqtadan keyin bo'shliq qo'yildi", 0.9)
        # N4: qo'shtirnoq normalizatsiyasi
        add(re.compile(r"[«„“]"), '"', "N4", "Qo'shtirnoq",
            "Qo'shtirnoq \"...\" ko'rinishiga keltirildi", 0.85)
        add(re.compile(r"»"), '"', "N4", "Qo'shtirnoq",
            "Qo'shtirnoq \"...\" ko'rinishiga keltirildi", 0.85)
        # N5: apostrof normalizatsiyasi
        if self.apostrophe == "unicode":
            add(re.compile(r"(?<=[oO])['\u02bb\u2018\u2019`]"), "\u02bb", "N5",
                "Apostrof", "o' harfi uchun imlo apostrofi (ʻ) qo'yildi", 0.8)
            add(re.compile(r"(?<=[gG])['\u02bc\u2018\u2019`]"), "\u02bc", "N5",
                "Apostrof", "g' harfi uchun imlo apostrofi (ʼ) qo'yildi", 0.8)
        add(re.compile(r"(?<=\w)[`\u00b4\u2018\u2019](?=\w)"), "'", "N5",
            "Apostrof", "Noto'g'ri apostrof belgisi to'g'rilandi", 0.9)
        if self.apostrophe == "ascii":
            add(re.compile(r"(?<=\w)[\u02bb\u02bc\u2019](?=\w)"), "'", "N5",
                "Apostrof", "Apostrof oddiy (') ko'rinishga keltirildi", 0.8)
        # N6: 4+ nuqta -> uch nuqta
        add(re.compile(r"\.{4,}"), "...", "N6", "Ko'p nuqta",
            "To'rt va undan ko'p nuqta uch nuqtaga keltirildi", 0.8)
        # N7: takroriy ?/!
        add(re.compile(r"\?{2,}"), "?", "N7", "Takror",
            "Takrorlangan so'roq belgilari bittaga keltirildi", 0.55)
        add(re.compile(r"!{2,}"), "!", "N7", "Takror",
            "Takrorlangan undov belgilari bittaga keltirildi", 0.55)
        # N8: qavs ichidagi bo'shliqlar
        add(re.compile(r"\([ \t]+"), "(", "N8", "Qavs",
            "Ochuvchi qavsdan keyingi bo'shliq olib tashlandi", 0.85)
        add(re.compile(r"[ \t]+\)"), ")", "N8", "Qavs",
            "Yopuvchi qavs oldidagi bo'shliq olib tashlandi", 0.85)
        # AV2: ikkilangan vergul
        add(re.compile(r",\s*,"), ",", "AV2", "Tozalash",
            "Ikkilangan vergul bittaga keltirildi", 0.8)
        # AV3: nuqtadan oldingi vergul / gap oxiridagi vergul
        add(re.compile(r",[ \t]*\."), ".", "AV3", "Tozalash",
            "Nuqtadan oldingi vergul olib tashlandi", 0.8)
        add(re.compile(r",[ \t]*$", re.M), ".", "AV3", "Tozalash",
            "Gap oxiridagi vergul nuqtaga almashtirildi", 0.78)
        # N9: en-dash -> tire
        add(re.compile(r"(?<=\s)–(?=\s)"), "—", "N9", "Tire",
            "Qisqa chiziqcha (–) tire (—) ga keltirildi", 0.5)
        return out

    def _final_edits(self, text: str, prot: List[Tuple[int, int]]) -> List[Edit]:
        """3-bosqich: qoldiq ikkilangan belgilar va bo'shliqlarni tozalash."""
        out: List[Edit] = []
        for m in re.finditer(r",[ \t]*,", text):
            out.append(Edit(m.start(), m.end(), ",", "AV2", "Tozalash",
                            "Ikkilangan vergul bittaga keltirildi", 0.8))
        for m in re.finditer(r"[ \t]+([,.;:!?…])", text):
            out.append(Edit(m.start(), m.end(), m.group(1), "N2", "Bo'shliq",
                            "Tinish belgisi oldidagi ortiqcha bo'shliq olib tashlandi", 0.95))
        for m in re.finditer(r"([,;:])(?=[^\s\d)])", text):
            out.append(Edit(m.start(), m.end(), m.group(1) + " ", "N3", "Bo'shliq",
                            "Tinish belgisidan keyin bo'shliq qo'yildi", 0.9))
        for m in re.finditer(r",[ \t]*\.", text):
            out.append(Edit(m.start(), m.end(), ".", "AV3", "Tozalash",
                            "Nuqtadan oldingi vergul olib tashlandi", 0.8))
        for m in re.finditer(r":[ \t]*:", text):
            out.append(Edit(m.start(), m.end(), ":", "I3", "Tozalash",
                            "Ikkilangan ikki nuqta bittaga keltirildi", 0.8))
        return out

    def _balance_checks(self, text: str) -> List[Edit]:
        """Qavs va qo'shtirnoq juftligi — faqat ogohlantirish (hech narsa o'zgartirmaydi)."""
        out: List[Edit] = []
        if text.count("(") != text.count(")"):
            n = max(text.count("("), text.count(")"))
            pos = max(text.rfind("("), text.rfind(")"))
            out.append(Edit(pos, pos, "", "P1", "Qavs",
                            "Qavslar muvozanati buzilgan (%s ochuvchi, %s yopuvchi) — "
                            "qo'lda tekshiring" % (text.count("("), text.count(")")), 0.9))
        if text.count('"') % 2 == 1:
            pos = text.rfind('"')
            out.append(Edit(pos, pos, "", "P1", "Qo'shtirnoq",
                            "Qo'shtirnoqlar juft emas — qo'lda tekshiring", 0.9))
        return out

    # ======================================================================
    # 10. T-QOIDALARI: GAP BOSHI VA OXIRI
    # ======================================================================

    def _r_T1(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """T1: bayon gap oxiriga nuqta."""
        term = s.terminator
        if term:                      # allaqachon nuqta/?/! bor
            return
        st = s.stripped
        # v7: qo'shtirnoq bilan tugagan gap - nuqta qo'shtirnoq ICHIGA
        # qo'yiladi (T7 hal qiladi): '...dedi.' emas, '... dedi."'
        if st.endswith('"') and len(st) >= 2:
            return
        words = self._words(toks)
        if not words:
            return
        q, _, _ = self._question_signal(s, words)
        # Dialog: "— Qalaysiz, — dedi u" — ? ichkariga, oxiriga nuqta kerak
        dialogue_author = (s.text.lstrip()[:1] in ("—", "–", '"', "«")
                           and re.search(r"[ \t][—–][ \t]", s.text))
        if q and not dialogue_author:      # so'roq gap — T2 hal qiladi
            return
        # Undov/salom so'zi bilan boshlangan QISQA gap — T3 egallaydi;
        # uzun gapda ("Assalomu alaykum, hurmatli ustoz, ...") nuqta kerak
        if words[0].lower in INTERJECTIONS or (
                words[0].lower in SALUTATIONS and len(words) <= 4):
            return
        has_fin = any(is_finite(w.lower) for w in words)
        conf = 0.86 if (has_fin or len(words) >= 3) else 0.55
        pos = s.start + len(s.stripped)
        out.append(Edit(pos, pos, ".", "T1", "Gap oxiri",
                        "Bayon gap oxiriga nuqta qo'yildi"
                        if conf >= 0.7 else
                        "Gap yoki sarlavha aniqlanmadi — nuqta tavsiya etiladi", conf))

    def _r_T2(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """T2: so'roq gap oxiriga so'roq belgisi (?)."""
        if s.terminator == "?":
            return
        # v7: qo'shtirnoq bilan TUGAGAN gapda belgi qo'shtirnoq ichiga
        # qo'yiladi (T7); qo'shtirnoq faqat gap ICHIDA bo'lsa T2 ishlaydi
        if s.stripped.endswith('"'):
            return
        words = self._words(toks)
        if not words:
            return
        q, why, conf = self._question_signal(s, words)
        if not q:
            if why and s.analysis.get("q_info"):
                qw = next((w for w in words if w.lower in QWORDS), words[0])
                out.append(Edit(qw.start, qw.start, "", "W5",
                                "Ma'lumot", why, 0.9))
            return
        term = s.terminator
        # DIALOG: "— Qalaysiz, — dedi u." — belgi gap oxiriga emas,
        # ko'chirma gapning o'z oxiriga (muallif tire'sidan oldingi vergulga) qo'yiladi
        lstripped = s.text.lstrip()
        if lstripped[:1] in ("—", "–", '"', "«") and re.search(r"[ \t][—–][ \t]", s.text):
            m = re.search(r",[ \t]*(?=[—–])", s.text)
            if m and s.terminator != "?":
                out.append(Edit(s.start + m.start(), s.start + m.end(), "? ",
                                "T2", "So'roq belgisi",
                                "Dialogdagi so'roq gap oxiriga so'roq belgisi "
                                "qo'yildi (muallif gapidan oldin)", 0.85))
            return
        pos = s.start + len(s.stripped)
        ends_ellipsis = s.stripped.endswith("...") or s.stripped.endswith("…")
        if term in (".", ",", "!") and not ends_ellipsis:
            out.append(Edit(pos - 1, pos, "?", "T2", "So'roq belgisi",
                            "So'roq gap oxiriga so'roq belgisi qo'yildi (%s)" % why,
                            min(0.9, conf + 0.02)))
        else:
            out.append(Edit(pos, pos, "?", "T2", "So'roq belgisi",
                            "So'roq gap oxiriga so'roq belgisi qo'yildi (%s)" % why,
                            min(0.9, conf + 0.02)))

    def _r_T3(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """T3: undov gap oxiriga undov belgisi (!).
        v7: buyruq mayli oxirida NUQTA (T1) qo'yiladi — undov belgisi faqat
        undov so'zi yoki salomlashish/minnatdorchilik bilan boshlangan gapda
        ("Esda saqlang:" ko'rsatmalari "!" bilan emas, "." bilan tugaydi)."""
        if s.terminator == "!":
            return
        if s.stripped.endswith('"'):   # nutq bilan tugagan gap - T7
            return
        words = self._words(toks)
        if not words:
            return
        q, _, _ = self._question_signal(s, words)
        if q:
            return
        fw = words[0].lower
        conf, why = 0.0, ""
        if fw in INTERJECTIONS:
            conf, why = 0.8, "gap boshida undov so'zi ('%s')" % words[0].text
        elif fw in SALUTATIONS and len(words) <= 4:
            conf, why = 0.74, "salomlashish/minnatdorchilik so'zi ('%s')" % words[0].text
        if not conf:
            # v7: buyruq mayli bilan tugagan gap — bayon gap (nuqta T1 dan)
            return
        term = s.terminator
        pos = s.start + len(s.stripped)
        ends_ellipsis = s.stripped.endswith("...") or s.stripped.endswith("…")
        if term in (".", ",") and not ends_ellipsis:
            out.append(Edit(pos - 1, pos, "!", "T3", "Undov belgisi",
                            "Undov gap oxiriga undov belgisi qo'yildi (%s)" % why, conf))
        elif not term:
            out.append(Edit(pos, pos, "!", "T3", "Undov belgisi",
                            "Undov gap oxiriga undov belgisi qo'yildi (%s)" % why, conf))

    def _r_T7(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """T7 (v7): to'g'ridan-to'g'ri nutq qo'shtirnoqlari.
        'U aytdi: "Bugun yomg'ir yog'adi." Men javob berdim: "Unda ..."' -
        nuqta/so'roq belgisi qo'shtirnoq ICHIGA, nutqdan keyingi gap bosh
        harf bilan, ikki nuqtadan keyingi nutq boshi bosh harf bilan."""
        text = s.text
        qpos = [m.start() for m in re.finditer(r'"', text)]
        if len(qpos) < 2 or len(qpos) % 2 != 0:
            return
        abs0 = s.start
        for qi in range(0, len(qpos), 2):      # ochuvchi qo'shtirnoq
            p = qpos[qi]
            inner = text[p + 1:qpos[qi + 1]] if qi + 1 < len(qpos) else ""
            # nutq boshi bosh harf bilan (ayniqsa ikki nuqtadan keyin)
            m = re.search(r"[^\W\d_]", inner)
            before = text[:p].rstrip()[-1:]
            if m and inner[m.start()].islower() and before in (":",):
                out.append(Edit(abs0 + p + 1 + m.start(),
                                abs0 + p + 1 + m.start() + 1,
                                inner[m.start()].upper(), "T7", "Bosh harf",
                                "To'g'ridan-to'g'ri nutq boshi bosh harf "
                                "bilan yozildi", 0.8))
        for qi in range(1, len(qpos), 2):      # yopuvchi qo'shtirnoq
            p = qpos[qi]
            q = p
            while q >= 1 and text[q - 1] in " \t":
                q -= 1                          # bo'shliqlarni yutamiz
            before = text[q - 1] if q >= 1 else ""
            if before and (before.isalnum() or before in "'ʻ’"):
                # nutq ichida tugatuvchi belgi yo'q: nutq turi aniqlanadi
                inner_words = re.findall(r"[\w'ʻ’-]+", text[qpos[qi - 1] + 1:q])
                ch = "."
                if inner_words and (inner_words[0].lower() in QWORDS
                        or any(self._mi_stem(x.lower(), loose=True) is not None
                               for x in inner_words)):
                    ch = "?"
                out.append(Edit(abs0 + q, abs0 + p, ch, "T7",
                                "So'roq belgisi" if ch == "?" else "Nutq nuqtasi",
                                "To'g'ridan-to'g'ri nutq oxiriga %s qo'shtirnoq "
                                "ichiga qo'yildi" % ("so'roq belgisi" if ch == "?"
                                else "nuqta"), 0.82))
            # yopuvchi qo'shtirnoqdan keyingi muallif gapi bosh harf bilan
            m = re.search(r"\S", text[p + 1:])
            if m:
                ch0 = text[p + 1:][m.start()]
                if ch0.isalpha() and ch0.islower():
                    out.append(Edit(abs0 + p + 1 + m.start(),
                                    abs0 + p + 1 + m.start() + 1,
                                    ch0.upper(), "T7", "Bosh harf",
                                    "Nutqdan keyingi muallif gapi bosh harf "
                                    "bilan yozildi", 0.85))

    def _r_T4(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """T4: har bir gap birinchi harfi bosh harfga."""
        m = re.search(r"[^\W\d_]", s.text)
        if not m:
            return
        i = m.start()
        ch = s.text[i]
        # Birinchi harfdan oldin faqat bo'shliq/qo'shtirnoq/qavs/tire turishi mumkin
        if i > 0 and re.search(r"[^\s\"'«([—–-]", s.text[:i]):
            return
        if ch.isupper():
            return
        # Dialog: "— so'radi u." — muallif gapi kichik harfda qoladi
        lstripped = s.text.lstrip()
        if lstripped[:1] in ("—", "–", "-"):
            words = self._words(toks)
            if words and is_finite(words[0].lower):
                return
        abs_pos = s.start + i
        out.append(Edit(abs_pos, abs_pos + 1, ch.upper(), "T4", "Bosh harf",
                        "Gap boshidagi so'z bosh harf bilan yozildi", 0.95))

    # ======================================================================
    # 11. V-QOIDALARI: VERGUL
    # ======================================================================

    def _intro_at_start(self, words: List[Tok]) -> Tuple[int, float]:
        """Gap boshidagi kirish so'zi/birikmasini topadi: (token soni, ishonch)."""
        if not words:
            return 0, 0.0
        lw = [w.lower for w in words[:3]]
        tri = " ".join(lw[:3])
        bi = " ".join(lw[:2])
        if tri in KIRISH_MULTI:
            return 3, KIRISH_MULTI[tri]
        if bi in KIRISH_MULTI:
            return 2, KIRISH_MULTI[bi]
        if lw[0] in KIRISH_STRONG:
            return 1, 0.93
        if lw[0] in ENUM_POSS:
            # v7: 'birinchisi, server barqarorligi; ikkinchisi, ...' —
            # sanash tartib so'zidan keyin vergul (keyingi so'z 'esa/ham'
            # bo'lmasa)
            if len(lw) > 1 and lw[1] not in ("esa", "ham", "yana"):
                return 1, 0.85
            return 0, 0.0
        if lw[0] in ORDINAL_KIRISH:
            return 1, 0.92
        if lw[0] in KIRISH_WEAK:
            return 1, 0.55
        if lw[0] == "aks" and len(lw) > 1 and lw[1] == "holda":
            return 2, 0.85
        if lw[0] == "shuning" and len(lw) > 1 and lw[1] == "uchun":
            return 2, 0.6   # gap boshida vergul ixtiyoriy — faqat tavsiya
        return 0, 0.0

    def _r_V1(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V1: gap boshidagi kirish so'zidan keyin vergul. "Afsuski, ...\" """
        words = self._words(toks)
        if len(words) < 2:
            return
        # Maxsus holat: "..., xolos." — gap oxirida, oldidan vergul
        if words[-1].lower == "xolos" and self._cb(words[-1]) != "," and len(words) > 1:
            out.append(Edit(words[-1].start, words[-1].start, ",", "V1", "Vergul",
                            "Cheklov ma'noli 'xolos' so'zidan oldin vergul qo'yildi", 0.8))
        n, conf = self._intro_at_start(words)
        if not n:
            return
        last = words[n - 1]
        if last is words[-1]:
            return
        phrase = " ".join(w.text for w in words[:n])
        e = self._comma_after(last, conf, "V1", "Vergul",
                              "Gap boshidagi kirish so'zi '%s' dan keyin vergul "
                              "qo'yildi" % phrase, toks)
        if e:
            out.append(e)

    def _r_V2(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V2: gap o'rtasidagi kirish so'zi ikki tomondan vergul bilan ajratiladi."""
        words = self._words(toks)
        if len(words) < 3:
            return
        for i in range(1, len(words) - 1):
            t = words[i]
            w = t.lower
            conf = 0.0
            enum_word = False
            if w in ENUM_POSS:
                # v7: 'birinchisi, server ...' — sanash tartib so'zidan keyin
                # vergul; oldinga vergul/nuqta-vergul QO'YILMAYDI (I5 beradi)
                if (words[i + 1].lower in ("esa", "ham", "yana")
                        or self._ca(t) == ","):
                    continue
                conf = 0.85
                enum_word = True
            elif w in KIRISH_STRONG or w in ORDINAL_KIRISH:
                # v7: 'haqiqatan ham' kabi ko'p so'zli birlik birinchi so'zi
                # bo'lsa — birlikdan KEYIN vergul (V1/izoh bilan), bu so'zning
                # o'zidan keyin emas
                if (i + 1 < len(words)
                        and (w + " " + words[i + 1].lower) in KIRISH_MULTI):
                    continue
                conf = 0.8
            elif w in KIRISH_WEAK:
                conf = 0.6
            if not conf:
                continue
            if self._cb(t) == "," or self._ca(t) == ",":
                continue
            # Oldingi so'z ham kirish so'zi bo'lmasin; 'iltimos qil-' qo'shma fe'l
            if words[i - 1].lower in KIRISH_STRONG:
                continue
            if w == "iltimos" and words[i + 1].lower.startswith("qil"):
                continue
            # Oldingi so'z bog'lovchi bo'lsa ("... lekin albatta keladi") —
            # faqat kirish so'zidan KEYIN vergul, oldinga emas
            prev_is_conj = (words[i - 1].lower in COMMA_TAKERS
                            or words[i - 1].lower in ENUM_CONJ)
            # "jumladan/masalan + ot" — birlik sifatida ajratamiz,
            # lekin birlikdan KEYIN ham ot/kesim ketsa ("masalan serverning
            # ishlash tezligi pasaydi") bu birlik emas — gap predmeti:
            # vergul faqat 'masalan' dan keyin qo'yiladi.
            unit_end = t
            if w in ("jumladan", "masalan") and i + 1 < len(words) \
                    and self._nounish(words[i + 1]):
                ue = i + 1
                if (i + 2 < len(words) and self._nounish(words[i + 2])
                        and not self._adjlike(words[i + 1])):
                    ue = i + 2
                if (ue + 1 >= len(words)
                        or self._nounish(words[ue + 1])
                        or is_finite(words[ue + 1].lower)):
                    ue = i
                unit_end = words[ue]
            e1 = None
            # "mutaxassislarning aytishicha" - genitive birlik: oldinga
            # vergul qo'yilmaydi (birlik buziladi)
            if (not prev_is_conj and not enum_word
                    and not words[i - 1].lower.endswith("ning")):
                e1 = self._comma_before(t, conf, "V2", "Vergul",
                                        "Kirish so'zi '%s' vergul bilan ajratildi"
                                        % t.text)
            e2 = self._comma_after(unit_end, conf - 0.03, "V2", "Vergul",
                                   "Kirish so'zi '%s' vergul bilan ajratildi" % t.text,
                                   toks)
            if e1:
                out.append(e1)
            if e2:
                out.append(e2)

    def _r_V15(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V15: "Ha", "Yo'q", "Qani", "Xo'sh" — bog'lanmagan so'zlardan keyin vergul;
        gap oxiridagi "to'g'rimi?" so'roq yuklamasi."""
        words = self._words(toks)
        if not words:
            return
        fw = words[0]
        if fw.lower in UNBOUND_WORDS and len(words) > 1:
            nxt = words[1]
            # "Nima qilding?" — faqat 'nima' + fe'l = so'roq boshi, vergul
            # shart emas. Qolganlari ('Ha tekshirdim') vergul TALAB etadi:
            # 'Ha, tekshirdim.' (v7.3)
            skip = fw.lower == "nima" and is_finite(nxt.lower)
            if not skip and self._ca(fw) != ",":
                out.append(Edit(fw.end, fw.end, ",", "V15", "Vergul",
                                "Bog'lanmagan '%s' so'zidan keyin vergul qo'yildi"
                                % fw.text, 0.86))
        # "Assalomu alaykum, ..." salomlashish formulasi
        if (fw.lower == "assalomu" and len(words) > 1
                and words[1].lower == "alaykum" and self._ca(words[1]) != ","):
            out.append(Edit(words[1].end, words[1].end, ",", "V15", "Vergul",
                            "Salomlashish formulasidan keyin vergul qo'yildi",
                            0.82))
        # Gap oxiridagi so'roq yuklamasi: "Kelasiz, to'g'rimi?"
        lw = words[-1]
        if len(words) >= 2 and lw.lower in QUESTION_TAGS and self._cb(lw) != ",":
            conf = 0.88
            out.append(Edit(lw.start, lw.start, ",", "V15", "Vergul",
                            "So'roq yuklamasi '%s' oldidan vergul qo'yildi" % lw.text,
                            conf))
            if s.terminator != "?":
                pos = s.start + len(s.stripped)
                ends_ellipsis = s.stripped.endswith("...") or s.stripped.endswith("…")
                if s.terminator in (".", ",") and not ends_ellipsis:
                    out.append(Edit(pos - 1, pos, "?", "V15", "So'roq belgisi",
                                    "So'roq yuklamasidan keyin so'roq belgisi qo'yildi",
                                    conf))
                elif not s.terminator or ends_ellipsis:
                    out.append(Edit(pos, pos, "?", "V15", "So'roq belgisi",
                                    "So'roq yuklamasidan keyin so'roq belgisi qo'yildi",
                                    conf))
        elif (len(words) >= 2 and lw.lower == "ha"
              and is_finite(words[-2].lower) and self._cb(lw) != ","):
            out.append(Edit(lw.start, lw.start, ",", "V15", "Vergul",
                            "Tasdiq yuklamasi 'ha' oldidan vergul qo'yildi", 0.8))

    def _r_V16(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V16: undov so'zlardan keyin vergul. "Voy, ...\" """
        words = self._words(toks)
        if len(words) < 2:
            return
        fw = words[0]
        if fw.lower not in INTERJECTIONS or self._ca(fw) == ",":
            return
        # "Ey do'stlar, ..." — keyingi so'z murojaat bo'lsa, vergul undan keyin (V3)
        if fw.lower in ("ey", "hey") and self._nounish(words[1]):
            return
        out.append(Edit(fw.end, fw.end, ",", "V16", "Vergul",
                        "Undov so'zi '%s' dan keyin vergul qo'yildi" % fw.text, 0.85))

    def _person_signal(self, words: List[Tok]) -> bool:
        """Gapda 1/2-shaxs/buyruq/so'roq belgisi bormi (murojaat aniqlash uchun)."""
        for w in words:
            if (w.lower in IMPERATIVE_WORDS
                    or any(w.lower.endswith(x) for x in
                           ("ngiz", "nglar", "sizlar", "ginsiz", "sangiz"))):
                return True
        # v7.3: 2-shaxs to'ldiruvchi olmosh shakllari
        # ('Hurmatli ustoz, sizga hisobotni yuboraman')
        if any(w.lower in ("sizga", "sizni", "sizning", "sizdan", "siz bilan")
               for w in words):
            return True
        # v7.3: 1-shaxs kesim (oxirgi fe'l): '... yuboraman', '... kutaman'
        for w in reversed(words):
            if is_finite(w.lower):
                lw = w.lower
                if lw.endswith(("man", "miz", "aman", "amiz", "dim", "dimiz")) \
                        and not lw.endswith("eman"):
                    return True
                break
        return False

    def _r_V3(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V3: murojaat (undalma) vergul bilan ajratiladi. "Do'stim, kelasizmi?\" """
        words = self._words(toks)
        n = len(words)
        if n < 2:
            return
        pers = self._person_signal(words)
        i = 0
        while i < n:
            t = words[i]
            w = t.lower
            is_voc = (w in MUROJAAT_STRONG or (w in MUROJAAT_KIN and pers)
                      or w in VOC_ADJ
                      or (w in KNOWN_UZ_NAMES and i + 1 < n
                          and words[i + 1].lower in SUBJECT_PRONOUNS))
            if not is_voc:
                i += 1
                continue
            # Murojaat ZANJIRI: "aziz do'stim", "hurmatli ustoz" - bitta
            # birlik: vergul faqat zanjir OXIRIGA qo'yiladi (orqasiga emas).
            ce = i
            if w in VOC_ADJ:
                for t2 in range(i + 1, min(n, i + 3)):
                    if words[t2].lower in VOC_HEADS:
                        ce = t2
                    else:
                        break
                if ce == i:
                    # sifat yonida murojaat bosh so'zi yo'q - "u aziz
                    # do'stimi" kabi holat: oddiy sifat, tegilmaydi
                    i += 1
                    continue
            unit_last = words[ce]
            unit_txt = " ".join(x.text for x in words[i:ce + 1])
            if i == 0:
                after = words[ce + 1].lower if ce + 1 < n else ""
                if ((w in VOC_ADJ and after in SUBJECT_PRONOUNS)
                        or (w in KNOWN_UZ_NAMES
                            and after in SUBJECT_PRONOUNS)
                        or pers or (
                        n >= 2 and not any(is_finite(x.lower)
                                           for x in words[1:]))):
                    e = self._comma_after(unit_last, 0.85, "V3", "Vergul",
                                          "Murojaat '%s' dan keyin vergul "
                                          "qo'yildi" % unit_txt, toks)
                    if e:
                        out.append(e)
                else:
                    out.append(Edit(t.end, t.end, "", "W3", "Shubha",
                                    "'%s' - murojaat yoki ega? Aniq emas, "
                                    "o'zgartirilmadi" % unit_txt, 0.55))
            elif ce == n - 1:
                prev = words[i - 1]
                if is_finite(prev.lower):
                    p = person_of(prev.lower)
                    if p in (1, 2):
                        e = self._comma_before(t, 0.85, "V3", "Vergul",
                                               "Gap oxiridagi murojaat oldidan "
                                               "vergul qo'yildi")
                        if e:
                            out.append(e)
                    elif p == 3:
                        out.append(Edit(t.start, t.start, "", "W3", "Shubha",
                                        "'%s' - murojaat yoki teskari ega? "
                                        "Aniq emas, o'zgartirilmadi" % unit_txt,
                                        0.55))
                elif (prev.lower in SALUTATIONS
                      or prev.lower in INTERJECTIONS
                      or prev.lower in GREETING_OPENERS):
                    e = self._comma_before(t, 0.75, "V3", "Vergul",
                                           "Gap oxiridagi murojaat '%s' oldidan "
                                           "vergul qo'yildi" % unit_txt)
                    if e:
                        out.append(e)
            else:
                prev = words[i - 1]
                if is_finite(prev.lower):
                    e1 = self._comma_before(t, 0.8, "V3", "Vergul",
                                            "Murojaat so'zi '%s' vergul bilan "
                                            "ajratildi" % unit_txt)
                    if e1:
                        out.append(e1)
                nxt = words[ce + 1] if ce + 1 < n else None
                if nxt is not None and nxt.lower not in COMMA_TAKERS:
                    # v7: vergul faqat MUROJAAT ekaniga dalil bo'lsa:
                    #   a) oldin salom/undov turadi ('Salom do'stim, ...',
                    #      'Assalomu alaykum, aziz do'stim, ...')
                    #   b) keyingi qismda 2-shaxs murojaat dalili bor
                    #      ('siz', buyruq mayli: '... do'stim, keling')
                    #   c) keyingi so'z olmosh ('do'stim, siz ...')
                    # Aks holda 'Shunda do'stim avval ... taklif qildi'
                    # dagi kabi so'z EGA bo'lib qoladi — vergul QO'YILMAYDI.
                    greeting = (prev.lower in SALUTATIONS
                                or prev.lower in INTERJECTIONS
                                or (i >= 2 and words[i - 2].lower in SALUTATIONS))
                    rest = words[ce + 1:ce + 6]
                    address = greeting or nxt.lower in SUBJECT_PRONOUNS or any(
                        x.lower in SUBJECT_PRONOUNS
                        or self._is_imperative_word(x.lower)
                        or x.lower.endswith(("ngiz", "siz", "nglar", "sizlar"))
                        for x in rest)
                    if address:
                        e2 = self._comma_after(unit_last, 0.8, "V3", "Vergul",
                                               "Murojaat so'zi '%s' vergul bilan "
                                               "ajratildi" % unit_txt, toks)
                        if e2:
                            out.append(e2)
                    else:
                        out.append(Edit(t.start, t.start, "", "W3", "Shubha",
                                        "'%s' - murojaat emas, ega bo'lishi "
                                        "ehtimoli yuqori — vergul qo'yilmadi"
                                        % unit_txt, 0.6))
            i = ce + 1
        # "Hurmatli Aziz," formulasi
        for i, t in enumerate(words[:-1]):
            if t.lower == "hurmatli" and words[i + 1].text[:1].isupper():
                name = words[i + 1]
                end_tok = name
                if (i + 2 < n and words[i + 2].lower in MUROJAAT_TITLES):
                    end_tok = words[i + 2]
                if self._ca(end_tok) != ",":
                    out.append(Edit(end_tok.end, end_tok.end, ",", "V3", "Vergul",
                                    "Murojaat formulasi '%s %s' dan keyin vergul qo'yildi"
                                    % (t.text, end_tok.text), 0.8))
        # v7.3: imzo formulasi: '... kutaman. Hurmat bilan, Aziz.' -
        # gap oxiridagi 'hurmat bilan <ism>' orasiga vergul
        for i in range(n - 2):
            if (words[i].lower == "hurmat" and words[i + 1].lower == "bilan"
                    and i + 2 == n - 1 and self._ca(words[i + 1]) != ","):
                out.append(Edit(words[i + 1].end, words[i + 1].end, ",", "V3",
                                "Vergul", "Imzo formulasi ('hurmat bilan "
                                "%s') orasiga vergul qo'yildi" % words[i + 2].text,
                                0.7))
        # v7: ism + "siz/sen" murojaati: "Azizbek, siz kecha yuborgan
        # hisobotni tekshirdingizmi?" - noma'lum ot (ism) gap boshida turib,
        # keyin 2-shaxs olmosh keladi -> vergul. Ma'lum ravishlar (ertaga,
        # bugun...) va fe'llar chiqarib tashlanadi.
        if n >= 3:
            w0, w1 = words[0], words[1]
            if (w0.lower not in FUNCTION_WORDS and w0.lower not in TIME_ADVERBS
                    and w0.lower not in KIRISH_STRONG and w0.lower not in ENUM_POSS
                    and self._nounish(w0) and not is_finite(w0.lower)
                    and w1.lower in ("siz", "sen")
                    and self._ca(w0) != ","):
                has_addr = (any(x.lower.endswith(("ngizmi", "sizmi", "ngiz", "siz"))
                                or x.lower in IMPERATIVE_WORDS
                                for x in words[2:8])
                            or s.terminator == "?"
                            or any(is_qword_at(words, k) for k in range(2, n)))
                if has_addr:
                    out.append(Edit(w0.end, w0.end, ",", "V3", "Vergul",
                                    "Murojaat (ism) '%s' dan keyin vergul "
                                    "qo'yildi" % w0.text, 0.8))

    def _r_V4(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V4: qarama-qarshilik bog'lovchisi oldidan vergul. "..., lekin ...\" """
        words = self._words(toks)
        for i in range(1, len(words)):
            t = words[i]
            w = t.lower
            conf = 0.0
            if w in CONTRAST_CONJ:
                conf = 0.9
            elif (w == "balki" and i >= 1
                  and (words[i - 1].lower in ("emas", "gina", "xolos")
                       or is_finite(words[i - 1].lower))):
                # "... emas, balki ..." / "... cheklanmaydi, balki ..."
                # v7.2: tugallangan kesimdan keyin ham vergul shart
                conf = 0.86
            elif w == "aks" and i + 1 < len(words) and words[i + 1].lower == "holda":
                conf = 0.85
            if not conf:
                continue
            if t is words[-1] or self._cb(t) == ",":
                continue
            e = self._comma_before(t, conf, "V4", "Vergul",
                                   "Qarama-qarshilik bog'lovchisi '%s' oldidan vergul "
                                   "qo'yildi" % t.text)
            if e:
                out.append(e)

    def _r_V5(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V5: sabab bog'lovchisi oldidan vergul. "..., chunki ...\" """
        words = self._words(toks)
        for i in range(1, len(words)):
            t = words[i]
            w = t.lower
            conf = 0.0
            if w in CAUSAL_CONJ:
                conf = 0.9
            elif w == "shuning" and i + 1 < len(words) and words[i + 1].lower == "uchun":
                conf = 0.86
            if not conf:
                continue
            if t is words[-1] or self._cb(t) == ",":

            e = self._comma_before(t, conf, "V5", "Vergul",

                                   % ("shuning uchun" if w == "shuning" else t.text))
            if e:
                out.append(e)

    def _r_V6(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V6: "ki" bog'lovchisi. Qoida: "ki" oldingi so'zga tutash yoziladi va
        undan keyin vergul qo'yiladi: "U aytdiki, biz g'olib bo'lamiz."
        (G'. Abdurahmonov darsligi: "Bilamanki, javoblar turlicha bo'ladi.")"""
        words = self._words(toks)
        for i in range(1, len(words) - 1):
            t = words[i]
            if t.lower != "ki":
                continue
            prev = words[i - 1]
            if prev.lower in ("chunki", "sababi", "zero", "balki", "ehtimolki",
                              "shubhasizki", "darhaqiqatki", "tabiiyki", "demakki"):
                continue
            nxt = words[i + 1]
            if is_finite(prev.lower) and self._cb(t) != ",":
                # tutash yozish + vergul
                out.append(Edit(prev.end, t.start, "", "V6", "Tutash yozish",
                                "'ki' bog'lovchisi oldingi so'zga tutash yozildi "
                                "va undan keyin vergul qo'yildi", 0.72))
                if self._ca(t) != ",":
                    out.append(Edit(t.end, t.end, ",", "V6", "Vergul",
                                    "'ki' bog'lovchisidan keyin vergul qo'yildi", 0.72))
            elif not is_finite(prev.lower) and self._cb(t) != ",":
                e = self._comma_before(t, 0.62, "V6", "Vergul",
                                       "'ki' bog'lovchisidan oldin vergul kerak bo'lishi "
                                       "mumkin — qo'lda tekshiring")
                if e:
                    out.append(e)
        # Tutash "-ki" shakllari: "aytdiki, ..." / "shubhasizki, ..." / "Balki, ..."
        for i, t in enumerate(words[:-1]):
            w = t.lower
            if not w.endswith("ki") or len(w) <= 3 or self._ca(t) == ",":
                continue
            # v7.3: '-ki' SIFAT uchun bog'lovchi EMAS: 'tizimning ichki
            # ma'lumotlari' — genetiv oldida attributive, vergul yo'q
            if (w in ("ichki", "tashqi", "oliki", "xoloski")
                    or (i >= 1 and words[i - 1].lower.endswith("ning"))):
                continue
            stem = w[:-2]
            if w in ("ehtimolki", "shubhasizki", "darhaqiqatki", "tabiiyki"):
                e = self._comma_after(t, 0.8, "V6", "Vergul",
                                      "'ki' bog'lovchisidan keyin vergul qo'yildi",
                                      words)
                if e:
                    out.append(e)
            elif len(stem) >= 3 and is_finite(stem):
                # aytdiki, bilamanki, sezdimki...
                e = self._comma_after(t, 0.72, "V6", "Vergul",
                                      "Tutash '-ki' bog'lovchisidan keyin vergul "
                                      "qo'yildi ('%s')" % t.text, words)
                if e:
                    out.append(e)
            elif len(stem) >= 3 and self._is_imperative_word(stem):
                # v7: unutmangki, eslangki... - buyruq mayli + '-ki'
                e = self._comma_after(t, 0.78, "V6", "Vergul",
                                      "Tutash '-ki' bog'lovchisidan keyin vergul "
                                      "qo'yildi ('%s')" % t.text, words)
                if e:
                    out.append(e)

    def _r_V7(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V7: "deb" so'zi oldidan vergul: "U keladi, deb aytdi."
        v7.5: vergul FAQAT qo'shtirnoq ichidagi ko'chirma gap oldidan
        qo'yiladi: '... "Men kelaman", deb aytdi'. Qo'shtirnoqsiz o'zlashtirma
        nutqda ('kelaman deb aytdi') vergul QO'YILMAYDI (rasmiy qoida).
        v7.2: 'deb' oldidagi so'z KESIM fe'l bo'lishi shart."""
        # qo'shtirnoq yo'q bo'lsa — umuman aralashmaymiz (xato vergul oldini olamiz)
        if '"' not in s.text and "\u00ab" not in s.text and "\u201c" not in s.text:
            return
        words = self._words(toks)
        for i in range(1, len(words)):
            t = words[i]
            if t.lower != "deb" or t is words[-1] or self._cb(t) == ",":
                continue
            if i + 1 < len(words) and words[i + 1].lower in (
                    "ataladi", "atalgan", "atalmish", "yoziladi", "belgilanadi",
                    "tanimoq", "bilish", "atasam"):
                continue
            # v7.2: oldingi so'z tugallangan kesim bo'lsin
            if not is_finite(words[i - 1].lower):
                continue
            e = self._comma_before(t, 0.72, "V7", "Vergul",
                                   "'deb' so'zi oldidan vergul qo'yildi")
            if e:
                out.append(e)

    def _r_V8(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V8: shart ergash gapidan keyin vergul. "Agar yomg'ir yotsa, chiqmayman." """
        words = self._words(toks)
        conj_tail = ("esa", "ham", "gina", "da", "hamda")
        # V8a: agar/agarda bilan boshlangan
        if words and words[0].lower in ("agar", "agarda", "basa"):
            for wi in range(1, len(words)):
                w = words[wi]
                if has_cond_suffix(w.lower):
                    if w is words[-1]:
                        break
                    # v7.5: shart-kesim + 'esa/ham/gina' bitta birlik:
                    # 'bo'lsa esa ...' — vergul kesim bilan BIRGA qo'yilmaydi
                    if (wi + 1 < len(words)
                            and words[wi + 1].lower in conj_tail):
                        break
                    e = self._comma_after(w, 0.82, "V8", "Vergul",
                                          "Shart ergash gapidan keyin vergul qo'yildi "
                                          "('%s')" % w.text)
                    if e:
                        out.append(e)
                    break
        # V8c: "... boramiz, agar poyezd chiqsa, ..." — agar gap o'rtasida
        for i in range(1, len(words)):
            t = words[i]
            if (t.lower in ("agar", "agarda") and i + 1 < len(words)
                    and is_finite(words[i - 1].lower) and self._cb(t) != ","):
                e = self._comma_before(t, 0.75, "V8", "Vergul",
                                       "Shart ergash gapidan oldin vergul qo'yildi")
                if e:
                    out.append(e)
                break
        # V8b: "agar"siz shart shakli
        for i in range(1, len(words) - 1):
            t = words[i]
            if not has_cond_suffix(t.lower):
                continue
            # v7.5: "bo'lsa ham", "bo'lsa esa" — bitta birlik, vergul yo'q
            if (i + 1 < len(words)
                    and words[i + 1].lower in ("ham", "esa", "gina")):
                continue
            if self._ca(t) == ",":
                continue
            e = self._comma_after(t, 0.7, "V8", "Vergul",
                                  "Shurt ergash gapidan keyin vergul qo'yildi ('%s')"
                                  % t.text)
            if e:
                out.append(e)

    def _r_V9(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V9: "garchi ... -sa-da" dan keyin vergul. "... tursa-da, tan olmadi.\" """
        words = self._words(toks)
        if not words or words[0].lower not in ("garchi", "garchand"):
            return
        for w in words[1:]:
            if re.search(r"(sa|ganda|gan)-da$", w.lower):
                if w is words[-1]:
                    break
                e = self._comma_after(w, 0.78, "V9", "Vergul",
                                      "'garchi ... -sa-da' konstruksiyasidan keyin "
                                      "vergul qo'yildi")
                if e:
                    out.append(e)
                return

    def _r_V12(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V12: ergash gaplar: "-gani uchun", "-gandan keyin", "-gach", "-ganda", "-ib"."""
        words = self._words(toks)
        for i, t in enumerate(words):
            w = t.lower
            # A: "-gani uchun" — ergash gap bosh gapdan KEYIN kelsa oldidan,
            #    gap boshida kelsa "uchun" dan keyin vergul
            if (w.endswith("gani") and i + 1 < len(words)
                    and words[i + 1].lower == "uchun"):
                if (i >= 1 and is_finite(words[i - 1].lower)
                        and self._cb(t) != ","):
                    e = self._comma_before(t, 0.72, "V12", "Vergul",
                                           "'-gani uchun' ergash gapidan oldin vergul "
                                           "qo'yildi")
                    if e:
                        out.append(e)
                elif (i == 0 and len(words) > 2
                        and self._ca(words[i + 1]) != ","):
                    e = self._comma_after(words[i + 1], 0.6, "V12", "Vergul",
                                          "'-gani uchun' ergash gapidan keyin vergul "
                                          "kerak bo'lishi mumkin", words)
                    if e:
                        out.append(e)
            # B: "-gandan keyin / so'ng"
            if (w.endswith("gandan") and i + 1 < len(words)
                    and words[i + 1].lower in ("keyin", "song", "keying")):
                if (i >= 1 and is_finite(words[i - 1].lower)
                        and self._cb(t) != ","):
                    e = self._comma_before(t, 0.72, "V12", "Vergul",
                                           "'-gandan keyin' ergash gapidan oldin "
                                           "vergul qo'yildi")
                    if e:
                        out.append(e)
                elif (i == 0 and len(words) > 2
                        and self._ca(words[i + 1]) != ","):
                    e = self._comma_after(words[i + 1], 0.6, "V12", "Vergul",
                                          "'-gandan keyin' ergash gapidan keyin "
                                          "vergul kerak bo'lishi mumkin", words)
                    if e:
                        out.append(e)
            # C: "-gach" ravishdoshidan keyin (Dadam kelgach, osh suzildi)
            if w.endswith("gach") and 0 < i < len(words) - 1:
                if (words[i + 1].lower not in ("ham", "esa")
                        and self._ca(t) != ","):
                    e = self._comma_after(t, 0.62, "V12", "Vergul",
                                          "'-gach' ravishdoshidan keyin vergul kerak "
                                          "bo'lishi mumkin ('%s')" % t.text)
                    if e:
                        out.append(e)
            # D: "-ganda" ravishdoshidan keyin
            if w.endswith("ganda") and 0 < i < len(words) - 1 and w not in COND_BLOCK:
                if (words[i + 1].lower not in ("ham", "esa")
                        and self._ca(t) != ","):
                    e = self._comma_after(t, 0.62, "V12", "Vergul",
                                          "'-ganda' ravishdoshidan keyin vergul kerak "
                                          "bo'lishi mumkin ('%s')" % t.text)
                    if e:
                        out.append(e)
            # E: "-ib" ravishdoshlari — v7: agar keyingi qismning O'Z
            # to'ldiruvchisi (obyekti) va kesimi bo'lsa, vergul qo'yiladi:
            # 'loyihani sinovdan o'tkazib, barcha kamchiliklarni aniqlashimiz
            # kerak'. Lekin fe'l birikmalari ('olib keldi', 'tayyorlab
            # qo'yishimiz', 'urinib ko'rish') BUTUNLAY tegilmaydi (S6).
            if w.endswith("ib") and 0 < i < len(words) - 1 and i + 1 < len(words):
                nxt = words[i + 1]
                nl = nxt.lower
                if (has_converb_suffix(w) and self._ca(t) != ","
                        and not is_vchain_aux(nl) and not has_converb_suffix(nl)
                        and nl not in FUNCTION_WORDS
                        and (nl in ("barcha", "hamma", "hech")
                             or nl.endswith(("ni", "larini", "mizni", "ngizni")))
                        and any(self._clause_end(x.lower) or x.lower in PRED_NOMINAL
                                or x.lower in COPULA_END
                                for x in words[i + 2:i + 12])):
                    e = self._comma_after(t, 0.78, "V12", "Vergul",
                                          "'-ib' ravishdoshi o'z to'ldiruvchisi "
                                          "bor keyingi qismdan vergul bilan "
                                          "ajratildi ('%s')" % t.text)
                    if e:
                        out.append(e)
                elif (has_converb_suffix(w) and nxt.lower.endswith("ib")
                        and self._ca(t) != ","):
                    out.append(Edit(t.end, t.end, "", "W4", "Shubha",
                                    "Uyushiq ravishdoshlar '-ib' vergul bilan "
                                    "ajratilishi mumkin ('U o'qib, yozib, tahlil qilib "
                                    "chiqdi'), lekin qo'shma fe'llar ('olib keldi') ham "
                                    "bor — o'zgartirilmadi", 0.4))

    def _r_V10(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V10: uyushiq bo'laklar (sanash). "Olma, nok va uzum oldim."
        Muhim: yakka 'va' oldida vergul QO'YILMAYDI (rasmiy qoida)."""
        words = self._words(toks)
        # --- A: "va/hamda/yoki" orqali bog'langan sanash
        for j, t in enumerate(toks):
            if not t.is_word or t.lower not in ENUM_CONJ or j == 0 or j == len(toks) - 1:
                continue
            prev = toks[j - 1]
            if not prev.is_word or self._ca(prev) == ",":
                continue
            if is_finite(prev.lower):
                continue  # qo'shma gap — vergul shart emas
            items_before = self._collect_items(toks, j - 1)
            item_after, after_next = self._after_item(toks, j + 1)
            if item_after is None:
                continue
            all_items = [t2 for t2, _ in reversed(items_before)] + [item_after]
            # v7.3: masdar birikmalari sanash EMAS: 'dam olish va sayr
            # qilish' — 'dam, olish' vergul NOTO'G'RI bo'ladi
            adj_items = [t2 for t2, _ in reversed(items_before)][-1:] \
                + [item_after]
            if any(masdar_like(x.lower) for x in adj_items if x):
                continue
            if after_next is not None and after_next.is_word:
                if self._nounish(after_next):
                    if self._similar2(item_after, after_next):
                        all_items.append(after_next)   # to'rtinchi bo'lak
                    # aks holda: umumiy bosh ot ("va Buxoro shaharlari") — aralashmaymiz
                elif not (is_finite(after_next.lower)
                          or has_converb_suffix(after_next.lower)):
                    continue  # tuzilish tushunarsiz — aralashmaymiz
            if len(all_items) < 3:
                # Sifat+ot juftliklari bo'yicha ogohlantirish
                self._v10_pairs_warning(words, out)
                continue
            conf = 0.82 if self._similar(all_items) else 0.58
            # Vergul faqat 'va'DAN OLDINGI bo'laklar orasiga qo'yiladi
            # (yakka 'va' oldida vergul qo'yilmaydi!)
            before_seq = [t2 for t2, _ in reversed(items_before)]
            for k in range(len(before_seq) - 1):
                a, b = before_seq[k], before_seq[k + 1]
                if "," in self._text[a.end:b.start]:
                    continue
                e = self._comma_before(b, conf, "V10", "Vergul",
                                       "Uyushiq bo'laklar orasiga vergul qo'yildi "
                                       "(yakka 'va' oldida vergul qo'yilmaydi)")
                if e:
                    out.append(e)
        # --- B: bog'lovchisiz sanash: "Non sut tuxum oldim"
        self._v10_nova(words, out)
        # --- C: ketma-ket uyushiq sifatlar: "ishonchli, xavfsiz, tushunarli va ..."
        self._v10_adjectives(words, out)
        # --- D (v7): (sifat+ot) JUFTLIKLARI sanashi:
        # "qizil olma, sariq nok va yashil uzum oldim"
        self._v10_pairs(words, out)

    def _v10_pairs_warning(self, words: List[Tok], out: List[Edit]) -> None:
        # Gapda umuman vergul bo'lmasaagina tavsiya beramiz (aks holda shovqin)
        if any(self._ca(w) == "," for w in words):
            return
        adj_noun_pairs = 0
        for i in range(len(words) - 1):
            if (self._adjlike(words[i]) and self._nounish(words[i + 1])
                    and self._ca(words[i]) != ","):
                adj_noun_pairs += 1
        if adj_noun_pairs >= 2:
            out.append(Edit(words[0].start, words[0].start, "", "W6", "Shubha",
                            "Sifat+ot juftliklari sanashi vergul talab qilishi mumkin "
                            "('qizil olma, sariq nok va yashil uzum') — "
                            "o'zgartirilmadi", 0.4))

    def _collect_items(self, toks: List[Tok], j: int) -> List[Tuple[Tok, str]]:
        """'va' dan chapga qarab uyushiq bo'laklarni yig'adi (vergul ajratgich ham
        hisobga olinadi). Natija: [(token, o'ngdagi ajratgich)] — eng yaqindan uzoqqa."""
        items: List[Tuple[Tok, str]] = []
        i = j
        while i >= 0 and len(items) < 7:
            t = toks[i]
            if t.is_word:
                if not self._nounish(t) or self._adjlike(t):
                    break
                # v7.2: genetiv ('komponentlarning o'lchami') - egalik
                # birikmaning boshi, sanash bo'lagi EMAS
                if t.lower.endswith("ning"):
                    break
                # "quyidagi shaharlar" bosh otini sanashga kiritmaymiz
                if i > 0 and toks[i - 1].is_word and toks[i - 1].lower in QUYIDA_SET:
                    break
                sep = (self._text[t.end:items[-1][0].start]
                       if items else " ")
                if items and not re.fullmatch(r",? +", sep):
                    break
                items.append((t, sep))
            elif t.text == "," and items:
                pass  # vergul — ajratgich, davom etamiz
            else:
                break
            i -= 1
        return items

    def _after_item(self, toks: List[Tok], j: int) -> Tuple[Optional[Tok], Optional[Tok]]:
        """'va' dan keyingi so'nggi bo'lak (sifat+ot bo'lsa juftlik)."""
        a = toks[j]
        if not a.is_word or not self._nounish(a):
            return None, None
        nxt = toks[j + 1] if j + 1 < len(toks) else None
        if (self._adjlike(a) and nxt is not None and nxt.is_word
                and self._nounish(nxt)):
            nn = toks[j + 2] if j + 2 < len(toks) else None
            return nxt, nn
        return a, nxt

    @staticmethod
    def _class_group(c: str) -> str:
        """Qo'shimcha klasslarni oilalarga birlashtirish (sanash o'xshashligi).
        'o'lchami, oralig'i va joylashuvini' - egalik '-i' va '-ini' bir oila."""
        if c in ("i", "ni", "ini", "si", "miz", "imiz"):
            return "poss"
        if c.startswith("lar"):
            return "plur"
        return c

    @staticmethod
    def _similar(items: List[Tok]) -> bool:
        if not items:
            return False
        # v7.4: HAMMASI KATTA HARF matnda (yoki bosh harf kirill matnda)
        # har so'z katta harf bilan boshlanadi — bu uyushiq dalili EMAS.
        # Kamida bitta bo'lakda davomiy kichik harflar bo'lsin (Title-case).
        if (all(t.text[:1].isupper() for t in items)
                and any(len(t.text) > 1 and t.text[1:] == t.text[1:].lower()
                        for t in items)):
            return True
        classes = {UzbekGrammar._class_group(end_class(t.lower))
                   for t in items}
        return len(classes) == 1

    @staticmethod
    def _similar2(a: Tok, b: Tok) -> bool:
        ca = UzbekGrammar._class_group(end_class(a.lower))
        cb = UzbekGrammar._class_group(end_class(b.lower))
        return ca == cb

    def _v10_nova(self, words: List[Tok], out: List[Edit]) -> None:
        """Bog'lovchisiz sanash: 'Non sut tuxum oldim' -> 'Non, sut tuxum oldim'.
        'hamma narsa', 'hech narsani' kabi qotib qolgan birliklar va
        postpozitsiyalar sanash bo'lagi hisoblanmaydi.

        v7.3: yakka 'va/hamda/yoki' ichida bo'lgan sanash ham aniqlanadi:
        'non sut tuxum va olma kirdi' -> 'non, sut, tuxum va olma kirdi'.
        Kesim/predikativ so'zlar ('bor', 'kerak', buyruq mayli) sanash
        bo'lagi hisoblanmaydi."""
        run: List[Tok] = []

        def flush(min_len: int, conf: float) -> None:
            items = list(run)
            # boshidagi qo'shimchali so'z sanash bo'lagi emas:
            # 'jamoada dasturchilar dizaynerlar va testlar ishlaydi'
            # (qisqa postpozitsiya/yuklama: 'da', 'ham' — tegilmaydi, v7.3)
            while (len(items) >= 3 and len(items[0].lower) >= 4
                   and items[0].lower.endswith(
                       ("da", "ga", "dan", "ning", "ni"))):
                items.pop(0)
            core = [x for x in items if x.lower not in ENUM_CONJ]
            if len(items) < min_len or len(core) < min_len - 1:
                return
            if not self._similar(core):
                return
            # v7.4: 'ko'p vaqt o'tar edi' — o'lchov/sifat + ot iborasi
            # sanash EMAS: qisqa runni o'lchov so'zi boshlagan bo'lsa,
            # bu birlik ibora (NP), uyushiq bo'laklar emas
            if (len(items) <= 3
                    and (items[0].lower in QUANTIFIER_WORDS
                         or self._adjlike(items[0]))):
                return
            for a, b in zip(items, items[1:]):
                if a.lower in ENUM_CONJ or b.lower in ENUM_CONJ:
                    continue          # yakka 'va' atrofida vergul yo'q
                if "," in self._text[a.end:b.start]:
                    continue
                e = self._comma_before(b, conf, "V10", "Vergul",
                                       "Uyushiq bo'laklar orasiga vergul "
                                       "qo'yildi")
                if e:
                    out.append(e)

        for i, t in enumerate(words):
            lw = t.lower
            # v7.4: kirill yozuvidagi so'zlar sanash runini uzadi — kirill
            # morfologiyasi tahlil qilinmaydi, end_class har doim "" qaytadi
            # va soxta 'uyushiq' vergul galvoni ishlab ketishi mumkin edi
            if CYRILLIC_RE.search(lw):
                run = []
                continue
            # v7.5: MIQDOR BIRLIKLARI ('125000 ta so'rov', '3 dona olma') —
            # birlik oldingi raqamga bog'lanadi, sanash runini uzadi
            if lw in ("ta", "dona", "foiz", "nafar", "marta", "yil", "oy",
                      "kun", "hafta", "soat", "daqiqa", "soniya", "ming",
                      "million", "mln", "mlrd", "sonli", "juft", "dasta",
                      "quti", "paket", "metr", "kilometr", "tonna"):
                run = []
                continue
            # v7.5: 'birma-bir', 'bir-bir' iboralari — sanash EMAS
            if lw in ("birma", "birma-bir", "bir-bir") or (
                    lw == "bir" and run
                    and run[-1].lower in ("birma", "bir", "bir-bir")):
                run = []
                continue
            # v7.5: FE'L SHAKLI runni uzadi: shart ('ajratsa') —
            # 'ajratsa asl mazmun' sanash EMAS, bosh qism kesimi
            if has_cond_suffix(lw) or (lw.endswith(("ib", "ab"))
                                       and not self._nounish(t)):
                if len(run) >= 3 and (is_finite(lw)
                                      or self._is_imperative_word(lw)
                                      or lw in PRED_NOMINAL):
                    flush(3, 0.7)
                run = []
                continue
            if lw in POSTPOSITIONS:
                continue                      # 'reja bo'yicha' - birlik
            if lw.endswith("ning"):
                # v7.2: genetiv ('komponentlarning o'lchami') - egalik
                # birikma boshlanadi, sanash runi uziladi
                run = []
                continue
            if lw in ENUM_CONJ:
                if run:
                    run.append(t)             # 'va' sanashni davom ettiradi
                continue
            if (self._nounish(t) and not self._adjlike(t)
                    and not self._is_imperative_word(lw)
                    and lw not in PRED_NOMINAL and lw not in COPULA_END):
                if i > 0 and (words[i - 1].lower, t.lower[:5]) in FROZEN_UNITS:
                    run = []                  # 'hamma narsa' - qotib qolgan
                    continue
                if run and not re.fullmatch(r" +", self._text[run[-1].end:t.start]):
                    run = [t]
                    continue
                run.append(t)
            else:
                if (len(run) >= 3
                        and (is_finite(lw) or self._is_imperative_word(lw)
                             or lw in PRED_NOMINAL)):
                    flush(3, 0.7)
                run = []
        flush(4, 0.6)

    def _r_V11(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V11: "ham ... ham" — ikkinchi (va uchinchi) 'ham' oldidan vergul.
        Shuningdek "yo ... yo" va "na ... na" takroriy bog'lovchilari.
        v7.5: 'haqiqatan ham', '... bilan ham', '... uchun ham' birliklari
        sanash EMAS — vergul QO'YILMAYDI."""
        words = self._words(toks)
        not_pairs_before = ("haqiqatan", "bilan", "kabi", "uchun",
                            "sifatida", "orqali", "hamda", "va", "gina")
        for conj in ("ham", "yo", "na"):
            idx = [i for i, w in enumerate(words) if w.lower == conj]
            if len(idx) < 2:
                continue
            conf = 0.85 if conj == "ham" else 0.8
            for k, i in enumerate(idx[1:], start=2):
                if i >= 1 and words[i - 1].lower in not_pairs_before:
                    continue          # birlik: 'haqiqatan ham', 'bilan ham'
                t = words[i]
                if self._cb(t) == ",":
                    continue
                e = self._comma_before(t, conf if k == 2 else conf - 0.05, "V11",
                                       "Vergul",
                                       "Takror qo'llanuvchi '%s' bog'lovchisi oldidan "
                                       "vergul qo'yildi" % t.text)
                if e:
                    out.append(e)

    def _r_V18(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V18: bog'lovchisiz qo'shma gap qismlari orasiga vergul.
        "Men bordim, u ham keldi." / "U ishlaydi, u o'qiydi.\" """
        words = self._words(toks)
        for i in range(len(words) - 2):
            t = words[i]
            if not is_finite(t.lower) or t.lower.endswith("gan"):
                continue
            # v7.5: belgi oldidan vergul — nutq ikki nuqtasi/vergul turgan joy
            if self._ca(t) in (",", ";", ".", "!", "?", "…", ":"):
                continue
            w1 = words[i + 1]
            subj = w1.lower in SUBJECT_PRONOUNS or (
                w1.text[:1].isupper() and w1.lower not in KIRISH_STRONG)
            if not subj:
                continue
            k = words[i + 2]
            if k.lower == "ham" and i + 3 < len(words):
                k = words[i + 3]
            if k is not None and is_finite(k.lower):
                out.append(Edit(t.end, t.end, ",", "V18", "Vergul",
                                "Bog'lovchisiz qo'shma gap qismlari orasiga vergul "
                                "qo'yildi", 0.75))

    def _r_V14(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V14: "ya'ni" atrofida vergul. "Biz, ya'ni yoshlar, hissa qo'shamiz.\" """
        words = self._words(toks)
        for i, t in enumerate(words):
            if t.lower not in ("yani",):
                continue
            if i >= 1 and self._cb(t) != ",":
                e = self._comma_before(t, 0.8, "V14", "Vergul",
                                       "Aniqlash ma'nosidagi 'ya'ni' oldidan vergul "
                                       "qo'yildi")
                if e:
                    out.append(e)
            if i < len(words) - 1:
                e = self._comma_after(t, 0.72, "V14", "Vergul",
                                      "Aniqlash ma'nosidagi 'ya'ni' dan keyin vergul "
                                      "qo'yildi", words)
                if e:
                    out.append(e)

    # 12. I-QOIDALARI: IKKI NUQTA
    # ======================================================================

    def _colon_in_gap(self, tok: Tok) -> bool:
        """Tokendan keyin (bo'shliqlardan so'ng) ikki nuqta bormi?"""
        j = tok.end
        while j < len(self._text) and self._text[j] in " \t":
            j += 1
        return j < len(self._text) and self._text[j] == ":"

    def _r_I1(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """I1: sanab ko'rsatishni kirishuvchi so'zlardan keyin ikki nuqta.
        "Quyidagi shaharlar: Toshkent, Samarqand va Buxoro.\" """
        words = self._words(toks)
        if not words:
            return
        for i, t in enumerate(words):
            if t.lower not in QUYIDA_SET:
                continue
            # Attributive/fe'liy ishlatilish: "quyidagi jadvalda ko'rsatilgan",
            # "quyidagi shartlarni bajaring: ..." (buyruq fe'li orqali
            # sanash kirishi kelgan holatda kolon fe'ldan KEYIN qo'yiladi
            # - I4 qoidasi, I1 bu yerda aralashmaydi; v7.1)
            if any(is_finite(w.lower) or self._is_imperative_word(w.lower)
                   for w in words):
                return
            head = None
            if t.lower == "quyidagi" and i + 1 < len(words) and self._nounish(words[i + 1]):
                head = words[i + 1]
                rest = words[i + 2:]
            else:
                rest = words[i + 1:]
            has_list = any(
                w.lower in ENUM_CONJ for w in rest) or sum(
                1 for w in rest if self._nounish(w)) >= 2
            if not has_list:
                return
            target = head if head is not None else t
            if self._colon_in_gap(target):     # ikki nuqta allaqachon bor
                return
            out.append(Edit(target.end, target.end, ":", "I1", "Ikki nuqta",
                            "Sanab ko'rsatishdan oldin ikki nuqta qo'yildi", 0.75))
            return

    def _r_I2(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """I2: to'g'ridan-to'g'ri nutqni keltiruvchi fe'ldan keyin ikki nuqta."""
        words = self._words(toks)
        if len(words) < 2:
            return
        for i, t in enumerate(words[:-1]):
            tail = self._text[t.end:s.end].lstrip()
            if tail[:1] != '"':
                continue
            # v7: har qanday kesim fe'l + qo'shtirnoq: 'berdim: "..."' ham
            is_speech = (t.lower in SPEECH_VERBS or is_finite(t.lower)
                         or (t.lower == "berdi" and i >= 1
                             and words[i - 1].lower == "javob"))
            if not is_speech:
                continue
            if self._colon_in_gap(t):
                continue
            # Keyinda qo'shtirnoq kelsa - ikki nuqta kerak (bir gapda bir
            # nechta nutq bo'lishi mumkin - davom etamiz)
            out.append(Edit(t.end, t.end, ":", "I2", "Ikki nuqta",
                            "To'g'ridan-to'g'ri nutqdan oldin ikki nuqta qo'yildi",
                            0.75))

    # 13. AV-QOIDALARI: ORTIQCHA BELGILARNI ANIQLASH
    # ======================================================================

    def _r_AV1(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """AV1: yakka 'va/hamda/yoki' oldidagi ortiqcha vergul.
        Qoida: teng bog'lovchi bir marta qatnashsa, vergul qo'yilmaydi."""
        for j, t in enumerate(toks):
            if not t.is_word or t.lower not in ENUM_CONJ or j == 0:
                continue
            if self._cb(t) != ",":
                continue
            comma = toks[j - 1]
            if not comma.text == ",":
                continue
            prev_word = None
            for k in range(j - 2, -1, -1):
                if toks[k].is_word:
                    prev_word = toks[k]
                    break
            next_word = None
            for k in range(j + 1, len(toks)):
                if toks[k].is_word:
                    next_word = toks[k]
                    break
            if prev_word is None or next_word is None:
                continue
            if self._nounish(prev_word) and self._nounish(next_word):
                # Sanashdagi ortiqcha vergul: "olma, va nok" -> "olma va nok"
                out.append(Edit(comma.start, t.start, "", "AV1", "Tozalash",
                                "Yakka '%s' oldidagi ortiqcha vergul olib tashlandi "
                                "(uyushiq bo'laklar vergulsiz bog'lanadi)" % t.text,
                                0.68))
            elif is_finite(prev_word.lower):
                # Qo'shma gapdagi vergul — odatda talab etilmaydi, lekin tuzilish
                # murakkab bo'lishi mumkin: faqat ogohlantiramiz
                out.append(Edit(comma.start, comma.start, "", "W7", "Shubha",
                                "O'zbek qoidasiga ko'ra yakki 'va' oldida vergul "
                                "odatda qo'yilmaydi — qo'lda tekshiring", 0.5))

    # 14. RUN-ON SEGMENTATSIYA: tinish belgisi YO'Q uzun matnlar (STT)
    #     uchun gap chegaralarini aniqlash.
    #     S1 - gap boshi formulasi oldidan nuqta
    #     S2 - to'g'ridan-to'g'ri so'roq gap: oldidan nuqta, oxiriga '?'
    #     S3 - bog'lovchisiz tutash gaplar: vergul yoki nuqta
    #     T9 - "Assalomu alaykum, aziz do'stim!" undov belgisi
    #     T10 - "Esingizda bo'lsin:" ikki nuqta formulasi
    # ======================================================================

    def _is_imperative_word(self, w: str) -> bool:
        """So'z buyruq mayli shaklidami ('aytingchi', 'saqlang'...)?"""
        if w in IMPERATIVE_WORDS:
            return True
        if w.endswith("ning"):
            # genitive: 'haftaning', 'buning', 'yozishning' - buyruq emas
            return False
        if is_finite(w):
            # 'ketding' kabi o'tgan zamon shakllari - buyruq emas
            return False
        for suf in ("ngchi", "nglar", "ginsiz", "gin", "ng", "sana"):
            if w.endswith(suf) and len(w) - len(suf) >= 3:
                # v7.5: '-ing' ko'pincha EGALIK qo'shimchasi ('dasturing',
                # 'xayoting', 'kitobing'); buyruq faqat fe'l-ildizdan
                # ('qiling', 'yozing', lug'atdagi shakllar)
                if w.endswith("ing") and w not in IMPERATIVE_WORDS:
                    stem = w[:-3]
                    if stem not in VERB_STEMS_MINI:
                        continue
                return True
        return False

    def _mi_stem(self, w: str, loose: bool = False) -> Optional[str]:
        """So'z '-mi' so'roq yuklamasi bilan tugasa - ildizini qaytaradi."""
        if not w.endswith("mi") or len(w) < 4:
            return None
        stem = w[:-2]
        if (is_finite(stem) or stem in MI_STEMS or stem in TIME_ADVERBS
                or stem in PRED_NOMINAL):
            return stem
        if loose and len(stem) >= 3 and stem not in MI_BLOCK:
            return stem
        return None

    def _clause_end(self, w: str) -> bool:
        """So'z gap qismini tugatadigan kesim/buyruq/so'roq shaklimi?"""
        if is_finite(w) or w in PRED_NOMINAL:
            return True
        if self._mi_stem(w) is not None:
            return True
        return self._is_imperative_word(w)

    def _scan_completer(self, words: List[Tok], start: int,
                        limit: int = 18) -> Optional[int]:
        """start indeksidan keyingi birinchi gap-tugatuvchini topadi."""
        for t in range(start, min(len(words), start + limit)):
            if self._clause_end(words[t].lower):
                if (t + 1 < len(words)
                        and words[t + 1].lower in OL_AUX):
                    return t + 1      # 'tayyorlay olamiz' - qo'shma kesim
                return t
        return None

    def _has_inner_punct(self, s: Sentence, toks: List[Tok],
                         prot: Optional[List[Tuple[int, int]]] = None) -> bool:
        """Gap ICHIDA (birinchi va oxirgi so'z oralig'ida) tinish belgisi
        bormi? Belgilar alohida token bo'lishi mumkin - shuning uchun
        butun oraliq matnida qidiriladi. Himoyalangan zonalar (09:30,
        3.14, sana, qisqartma) hisobga olinmaydi."""
        words = [t for t in toks if t.is_word]
        if len(words) < 2:
            return False
        lo, hi = words[0].end, words[-1].start
        chars = list(self._text[lo:hi])
        if prot:
            for ps, pe in prot:
                a, b = max(ps, lo), min(pe, hi)
                if a < b:
                    for x in range(a - lo, b - lo):
                        chars[x] = " "
        inner = "".join(chars)
        # v7.2: QO'SHTIRNOQLAR («» "" '') va TIRE (— –) endi bloklamaydi.
        # STT matnida yakka qo'shtirnoq yoki tire uchrashi mumkin, lekin
        # gap chegaralari baribir kerak (E7 testida 5000 belgili yozuv
        # bitta qo'shtirnoq tufayli umuman ajratilmay qolgandi).
        # Faqat HAQIQIY tinish belgilari (vergul/colon/tutuq/so'roq)
        # bo'lsa 'allaqachon punktuatsiyalangan' deb hisoblanadi.
        # v7.3: boshidagi SARLAVHA koloni ('Dolzarb: ...') - kolon faqat
        # birinchi so'zdan keyin turib, boshqa belgi bo'lmasa, bu sarlavha
        # hisoblanadi va gap chegaralarini BLOKLAMAYDI (N8 matni kabi
        # 20+ so'zli run-on ajratilishi kerak).
        punct_spans = list(re.finditer(r"[,;:!?…()\[\]{}]", inner))
        if punct_spans:
            if (len(punct_spans) == 1 and punct_spans[0].group() == ":"
                    and lo + punct_spans[0].start() <= words[0].end + 1
                    and len(words) >= 8):
                return False
            return True
        return False

    def _starter_match(self, words: List[Tok], j: int) -> Tuple[float, str]:
        """words[j] dan boshlanuvchi gap-boshi formulasi:
        (ishonch, mos kelgan so'zli formula) qaytaradi.
        v7: ko'p so'zli kirish birikmalari ('shunday ekan', 'haqiqatan
        ham', 'xulosa qilib aytganda') ham yangi gap boshlay oladi."""
        lw = [w.lower for w in words[j:j + 3]]
        for k in (3, 2):
            if len(lw) >= k and " ".join(lw[:k]) in SENT_START_MULTI:
                return SENT_START_MULTI[" ".join(lw[:k])], " ".join(lw[:k])
        # v7: kirish birikmasi - yangi gap boshi ('aks holda' bundan mustasno:
        # u ko'pincha shart-ogohlantirish sifatida vergul bilan ajratiladi)
        for k in (3, 2):
            if len(lw) >= k:
                ph = " ".join(lw[:k])
                if ph in KIRISH_MULTI and ph != "aks holda":
                    return KIRISH_MULTI[ph], ph
        w0 = lw[0]
        if w0 in SENT_START_STRONG:
            return 0.78, w0
        if w0 in ORDINAL_KIRISH:
            return 0.76, w0
        if w0 in SENT_START_TEMPORAL:
            return 0.75, w0
        if w0 in KIRISH_STRONG:
            return 0.73, w0
        if w0 in SENT_START_WEAK:
            return 0.70, w0
        return 0.0, ""

    def _question_stretch(self, words: List[Tok], j: int):
        """words[j..] bilan boshlanuvchi qism to'g'ridan-to'g'ri so'roq gapmi?
        Qaytaradi: (oxirgi so'z indeksi, turi) yoki None.
        '-mi' so'zlar tanlov varianti hisoblanadi va skanni TO'XTATMAYDI."""
        k = None
        mis: List[int] = []
        for t in range(j, min(len(words), j + 18)):
            w = words[t].lower
            if is_complement(w) or w in KNOW_VERBS or w in ("degan", "deb"):
                return None          # qaram (ergash) gap - so'roq emas
            if self._mi_stem(w, loose=True) is not None:
                mis.append(t)        # tanlov varianti davom etadi
                continue
            if self._clause_end(w):
                if (t + 1 < len(words)
                        and words[t + 1].lower in OL_AUX):
                    k = t + 1
                else:
                    k = t
                break
        stop = k + 1 if k is not None else min(len(words), j + 18)
        stretch = [w.lower for w in words[j:stop]]
        has_qw = any(is_qword_at(words, t) for t in range(j, stop))
        # v7.5: birinchi so'roq so'zining o'rni — bo'lak BOSHIDA bo'lishi kerak:
        # 'Bugun kim keldi?' ✓ / '... qanday munosabatda ishlashini ham
        # tekshirish zarur' — so'roq so'zi o'rtada, bu bayon gap
        qw_idx = None
        for t in range(j, stop):
            if is_qword_at(words, t):
                qw_idx = t
                break
        if len(mis) >= 2 and (has_qw or "yoki" in stretch or k is None):
            # ketma-ket '-mi' variantlar orasidagi masofa <= 5 bo'lsin -
            # katta oraliq bitta tanlov so'rogi emas, alohida so'roqlar
            if any(b - a > 5 for a, b in zip(mis, mis[1:])):
                return None
            # oxirgi '-mi' dan keyin qisqa fe'l qismi kelsa ('... indinmi
            # ketamiz?') - so'rog' fe'lgacha davom etadi
            if k is not None and k > mis[-1] and k - mis[-1] <= 3:
                return k, "alt"
            return mis[-1], "alt"
        if (has_qw and k is not None and k >= j + 2
                and qw_idx is not None and qw_idx - j <= 1):
            return k, "qw"
        return None

    def _alt_commas(self, words: List[Tok], j: int, k: int, out: List[Edit],
                    emit=None) -> None:
        """Tanlov so'rog'i variantlari orasiga vergul:
        'Bugunmi, ertami yoki indinmi?' (yoki oldida vergul yo'q)."""
        mis = [t for t in range(j, k + 1)
               if self._mi_stem(words[t].lower, loose=True) is not None]
        for a, b in zip(mis, mis[1:]):
            between = [w.lower for w in words[a + 1:b]]
            if any(x in ENUM_CONJ for x in between):
                continue
            if b - a > 5:
                continue
            w = words[a]
            if self._ca(w) == ",":
                continue
            if emit is not None:
                emit(w.end, ",", "S3", "Vergul",
                     "Tanlov so'rogining variantlari orasiga vergul "
                     "qo'yildi", 0.75)
            else:
                out.append(Edit(w.end, w.end, ",", "S3", "Vergul",
                                "Tanlov so'rogining variantlari orasiga "
                                "vergul qo'yildi", 0.75))

    def _voc_chain_end(self, words: List[Tok], start: int) -> Optional[int]:
        """Murojaat zanjirining oxirgi indeksi: 'aziz do'stim' -> do'stim."""
        end = None
        for t in range(start, min(len(words), start + 3)):
            w = words[t].lower
            if w in VOC_HEADS:
                end = t
            elif w in VOC_ADJ and end is None:
                continue
            else:
                break
        return end

    def _asyndetic(self, words: List[Tok], j: int, prev_len: int):
        """Bog'lovchisiz tutash mustaqil gap qismlari: vergul yoki nuqta."""
        prev_w = words[j - 1].lower
        # v7.3: 'shart bajarilmasa, ...' - ot-kesim so'zi + shart fe'li
        # bitta birlik (ot - ega), chegara QO'YILMAYDI
        if prev_w in PRED_NOMINAL and has_cond_suffix(words[j].lower):
            return None
        if (prev_w.endswith("gan")
                and words[j].lower not in SUBJECT_PRONOUNS
                and words[j].lower not in DEMONSTRATIVES):
            # 'rejalashtirib yurgan muhim loyiha' - sifat-ravishdosh,
            # kesim emas: chegara qo'ymaymiz. LHV keyingi so'z olmosh
            # bo'lsa ('yuz bermagan, bu juda katta yutuq') - kesim.
            return None
        # v7.2: nutq-keltiruvchi birlik '"X" degan xabar' - vergul/chegara
        # qo'ymaymiz (qo'shtirnoq ichidagi namuna gapning davomidir)
        if words[j].lower in ("degan", "deb", "degani", "deganlar"):
            return None
        if j + 1 < len(words) and words[j + 1].lower in ENUM_CONJ:
            # "dedi u va qo'lini cho'zdi" - bog'lovchi darhol kuzatilgan:
            # yangi mustaqil gap emas
            return None
        k = None
        for t in range(j + 1, min(len(words), j + 24)):
            # ICHKIRI bog'lovchi ('... parollarni YOKI shaxsiy ...'
            # sanash) to'siq emas - faqat words[j] dagi bog'lovchi D-shox
            # tomonidan ushlanadi
            tw = words[t].lower
            if (tw in COPULA_END and t >= j + 2
                    and (masdar_like(words[t - 1].lower)
                         or words[t - 1].lower.endswith(
                             ("ishi", "ligi", "ligini", "gani")))):
                # 'tugatish emas' - masdar + kopula bitta birlik (S6)
                continue
            if self._clause_end(tw):
                # v7.5: kesim + vchain-yordamchi faqat RAVIShDOSH-kesimdan
                # keyin ('ishlay boshladi'); 'bajarildi qolgan' kabi holatda
                # 'qolgan' yangi gapning ega-qismi bo'lishi mumkin
                if t + 1 < len(words) and (
                        words[t + 1].lower in OL_AUX
                        or (is_vchain_aux(words[t + 1].lower)
                            and (is_converb_like(tw)
                                 or tw.endswith(("ib", "ay"))))):
                    k = t + 1      # 'ishlay boshladi' - qo'shma kesim (S6)
                else:
                    k = t
                break
        if k is None:
            return None
        next_len = k - j + 1
        if next_len < 2:
            return None
        if self._is_imperative_word(words[k].lower):
            if has_cond_suffix(prev_w):
                # 'agar ... kelmasa, oldindan xabar bering' - shart ergash
                # gapidan keyin buyruq - NUQTA emas, VERGUL
                return ",", 0.74, ("Shart ergash gapidan keyin vergul "
                                   "qo'yildi")
            return ".", 0.74, ("Keyingi qism buyruq maylida - mustaqil gap "
                               "sifatida nuqta bilan ajratildi")
        if next_len >= 9:
            return ".", 0.72, ("Tutash qismlar juda uzun - mustaqil gaplar "
                               "sifatida nuqta bilan ajratildi")
        if prev_len >= 8:
            return ".", 0.72, ("Oldingi gap qismi uzun - yangi gap boshlandi")
        return ",", 0.72, ("Bog'lovchisiz qo'shma gap qismlari orasiga vergul "
                           "qo'yildi")

    def _right_clause_len(self, words: List[Tok], j: int, limit: int = 16) -> int:
        """words[j..] qismida birinchi kesimgacha bo'lgan so'z soni.
        0 - o'ngda mustaqil kesim topilmadi."""
        for t in range(j, min(len(words), j + limit)):
            if self._clause_end(words[t].lower):
                return t - j + 1
        return 0

    def _right_has_predicate(self, words: List[Tok], j: int,
                             limit: int = 16) -> bool:
        return self._right_clause_len(words, j, limit) > 0

    def _is_pron_form(self, w: str) -> bool:
        """v7.3: olmoshning qo'shimchali shakllari ('uni', 'unga', 'buni',
        'shunga'...) — yangi bo'lak boshi bo'la oladi (ega tushib qolgan)."""
        for suf in ("ni", "ga", "dan", "ning", "da", " bilan"):
            if w.endswith(suf):
                stem = w[:-len(suf)]
                if stem in (SUBJECT_PRONOUNS | DEMONSTRATIVES):
                    return True
        return False

    def _agar_leads(self, words: List[Tok], j: int) -> bool:
        """v7.3: words[j] dagi 'agar' ergash gapidan KEYIN asosiy qism
        bormi? ('agar tarmoq uzilsa, zaxira serverga o'tamiz' - bor;
        '..., agar kerak bo'lsa' - yo'q). Bor bo'lsa 'agar' yangi gapni
        boshlaydi (nuqta), yo'q bo'lsa oxirgi bo'lak (vergul).
        v7.5: ergash qism shart-kesimda ('... qilsa') tugashi va undan
        keyin haqiqiy kesim qidiriladi ('agar dastur ... qabul qilsa,
        kerakli ... joylashtira olmaydi' — bor)."""
        n = len(words)
        aux_after = {norm(x) for x in ("bo'lsa", "bo'lmasa", "bo'lsa-da",
                                       "ham", "esa", "gina", "emas",
                                       "edi", "ekan", "bo'ladi",
                                       "bo'lmaydi", "yana", "faqat",
                                       "hatto", "hamda", "va", "yoki",
                                       "ammo", "lekin", "biroq")}
        # 1) ergash qism: 'agar' dan keyingi birinchi shart-kesim
        sub_end = None
        t = j + 1
        while t < min(n, j + 16):
            w2 = words[t].lower
            if w2 in ("agar", "agarda"):
                return False          # ichma-ich shartlar — aralashmaymiz
            if has_cond_suffix(w2):
                sub_end = t
                break
            if self._clause_end(w2):
                nxt2 = words[t + 1].lower if t + 1 < n else ""
                if nxt2 in aux_after or nxt2 in POSTPOSITIONS:
                    t += 2            # kesim + yordamchi — davom etyapti
                    continue
                if nxt2 and has_cond_suffix(nxt2):
                    # 'yaratmoqchi bo'lsak' — qoshma shart kesim birligi
                    sub_end = t + 1
                    break
                return False          # ergash qismi yo'q — 'agar' oxirgi bo'lak
            t += 1
        if sub_end is None:
            return False
        # 2) asosiy qism: shart-kesimdan keyin kesim bo'lishi kerak
        for t in range(sub_end + 1, min(n, sub_end + 22)):
            w2 = words[t].lower
            if w2 in ("agar", "agarda"):
                return False
            if self._clause_end(w2):
                return True
        return False

    def _quote_pass_edits(self, text: str) -> List[Edit]:
        """v7 (1.4-bosqich): NUTQ FAZASI - qo'shtirnoqli matnni run-on
        segmentatsiyasidan OLDIN tartibga soladi:
          - 'aytdi "' -> 'aytdi: "' (I2)
          - 'yog\'adi "' -> 'yog\'adi."' (nutq nuqtasi qo'shtirnoq ichiga)
          - ikki nuqtadan keyingi nutq boshi bosh harf bilan
        Shundan keyin gap bo'luvchi '.'+qo'shtirnoqni tanib, qolgan
        qismlarni odatiy segmentator ishlaydi."""
        edits: List[Edit] = []
        qpos = [m.start() for m in re.finditer(r'"', text)]
        if len(qpos) < 2 or len(qpos) % 2 != 0:
            return edits
        for qi in range(0, len(qpos), 2):          # ochuvchi qo'shtirnoq
            op, cl = qpos[qi], qpos[qi + 1]
            # oldingi so'z + ikki nuqta
            j = op - 1
            while j >= 0 and text[j] in " \t":
                j -= 1
            wend = j + 1
            wstart = j
            while wstart >= 0 and re.match(r"[\w'\u02bb\u2019-]", text[wstart]):
                wstart -= 1
            wstart += 1
            word = norm(text[wstart:wend]) if wstart < wend else ""
            gap = text[wend:op]
            has_colon = ":" in gap
            if (word and (is_finite(word) or word in SPEECH_VERBS)
                    and not has_colon):
                edits.append(Edit(wend, wend, ":", "I2", "Ikki nuqta",
                                  "To'g'ridan-to'g'ri nutqdan oldin ikki nuqta "
                                  "qo'yildi ('%s')" % word, 0.75))
                has_colon = True
            # nutq boshi bosh harf bilan (faqat ikki nuqtadan keyin)
            if has_colon:
                m = re.search(r"[^\W\d_]", text[op + 1:cl])
                if m and text[op + 1 + m.start()].islower():
                    edits.append(Edit(op + 1 + m.start(), op + 2 + m.start(),
                                      text[op + 1 + m.start()].upper(), "T7",
                                      "Bosh harf",
                                      "To'g'ridan-to'g'ri nutq boshi bosh harf "
                                      "bilan yozildi", 0.8))
        for qi in range(1, len(qpos), 2):          # yopuvchi qo'shtirnoq
            p = qpos[qi]
            q = p
            while q >= 1 and text[q - 1] in " \t":
                q -= 1
            before = text[q - 1] if q >= 1 else ""
            if before and (before.isalnum() or before in "'\u02bb\u2019"):
                # v7.2: qo'shtirnoqdan keyin 'degan/deb/...' kelsa - bu
                # gap ICHIDAGI namuna ('"X" degan xabar') - oxirgi belgi
                # QO'YILMAYDI (etalon: "Parol noto'g'ri kiritildi. Qayta
                # urinib ko'ring" degan xabar ...)
                m2 = re.match(r"\s*([\w'\u02bb\u2019-]+)", text[p + 1:])
                nxt_word = norm(m2.group(1).lower()) if m2 else ""
                if nxt_word in ("degan", "degani", "deganlar", "deb", "dedi",
                                "dedilar", "deydi", "derdi", "deganmiz",
                                "deganingiz", "deganimda"):
                    continue
                inner_words = re.findall(r"[\w'\u02bb\u2019-]+",
                                         text[qpos[qi - 1] + 1:q])
                ch = "."
                if inner_words and (inner_words[0] in QWORDS
                        or any(self._mi_stem(x, loose=True) is not None
                               for x in inner_words)):
                    ch = "?"
                edits.append(Edit(q, p, ch, "T7",
                                  "So'roq belgisi" if ch == "?" else "Nutq nuqtasi",
                                  "To'g'ridan-to'g'ri nutq oxiriga %s qo'shtirnoq "
                                  "ichiga qo'yildi" % ("so'roq belgisi"
                                  if ch == "?" else "nuqta"), 0.82))
        return edits

    def _segment_runon(self, s: Sentence, stoks: List[Tok],
                       prot: Optional[List[Tuple[int, int]]] = None
                       ) -> List[Edit]:
        """Tinish belgisi YO'Q uzun gapni tahlil qilib, gap chegaralarini
        (nuqta / so'roq belgisi / vergul) taklif qiladi. Faqat ishonchli
        lingvistik dalil bo'lganda chegara qo'yadi; shubhali joylarga
        tegilmaydi (xavfsizlik saqlanadi)."""
        words = self._words(stoks)
        # v7.4: salomlashuv formulasi bilan boshlanuvchi 3 so'zlik gapga ham
        # ruxsat: 'xayr men ketdim' -> 'Xayr! Men ketdim.'
        greeting_short = (len(words) == 3
                          and words[0].lower in GREETING_OPENERS)
        if ((len(words) < 4 and not greeting_short)
                or self._has_inner_punct(s, stoks, prot)):
            return []
        # dialog satri ("— dedi u") - muallif gapi konventsiya bilan
        # punktuatsiyalanadi, aralashmaymiz
        if s.text.lstrip()[:1] in ("—", "–", "-"):
            return []
        out: List[Edit] = []
        emitted: List[int] = []      # belgi qo'yilgan pozitsiyalar
        caps: List[int] = []         # bosh harfga o'tkazilgan so'z boshlari
        n = len(words)

        def has_emit(pos: int) -> bool:
            return pos in emitted

        def cap_next(idx: int) -> None:
            if idx >= n:
                return
            w = words[idx]
            if w.start in caps:
                return
            caps.append(w.start)
            if not w.text[:1].isalpha():
                return
            ch = w.text[0]
            if ch.islower():
                out.append(Edit(w.start, w.start + 1, ch.upper(), "T4",
                                "Bosh harf", "Yangi gap boshidagi so'z bosh "
                                "harf bilan yozildi", 0.95))

        def emit(pos: int, ch: str, rule: str, name: str, msg: str,
                 conf: float) -> None:
            if pos in emitted:
                return
            # matnda bu pozitsiyada allaqachon xuddi shu belgi tursa
            # (qayta ishlov - idempotentlik) qayta qo'ymaymiz
            if pos < len(self._text) and self._text[pos] == ch:
                return
            # v7.3: o'sha joyda BOSHQA tinish belgisi tursa ham qo'ymaymiz
            # (masalan, sarlavha koloni oldiga nuqta QO'YILMAYDI)
            if pos < len(self._text) and self._text[pos] in ".,;:!?…":
                return
            out.append(Edit(pos, pos, ch, rule, name, msg, conf))
            emitted.append(pos)

        # --- v7.5: QATTIQ LINGVISTIK HIMOYALAR (xato chegaralarni bloklaydi)
        def left_complete(cs: int, i: int) -> bool:
            """Chap segmentda haqiqiy kesim bo'lishi kerak: kesim so'zi,
            masdar-kesim, 'ega' yoki predikativ sifat. Kesimsiz qismdan
            ('men odatdagidan ertaroq') keyin chegara QO'YILMAYDI — chunki
            keyingi fe'l shu qismining kesimi bo'lishi mumkin."""
            if i < cs:
                return False
            for t in range(cs, i + 1):
                if self._clause_end(words[t].lower):
                    return True
            lw2 = words[i].lower
            if masdar_like(lw2) or lw2 == "ega" or lw2 in PRED_ADJ:
                return True
            return False

        def unresolved_qword(cs: int, j: int) -> bool:
            """Chap qismda YOPILMAGAN so'roq so'zi bormi? So'roq so'zidan
            keyin chegaraqidacha kesim ham, 'deb/degan' ham bo'lmasa — qaram
            nutq hali davom etyapti ('nega dasturing hali ham xato qilmoqda
            deb soradi' — chegara QO'YILMAYDI)."""
            for t in range(cs, j):
                if is_qword_at(words, t):
                    if any(self._clause_end(words[u].lower)
                           for u in range(t + 1, j)):
                        continue      # kesim bilan yopilgan
                    if any(words[u].lower in ("deb", "degan", "degani",
                                              "deganda", "dediki")
                           for u in range(t + 1, j)):
                        continue      # o'zlashtirma bilan yopilgan
                    return True
            return False

        def degan_frame_ahead(j: int, limit: int = 14) -> bool:
            """O'ng tomonda '... degan/deb + kesim' nutq-ramkasi bormi?
            ('agar vaqtim bolsa ... uchrashaman degan matnni yubordi' —
            bu bitta gap, 'agar' oldidan nuqta QO'YILMAYDI)."""
            for t in range(j, min(n, j + limit)):
                if words[t].lower in ("degan", "degani", "deb", "dediki"):
                    for u in range(t + 1, min(n, t + 4)):
                        if is_finite(words[u].lower):
                            return True
                    return False
            return False

        # --- T10: "Esingizda bo'lsin:" / "Esda saqlang:" - ikki nuqta
        head2 = " ".join(w.lower for w in words[:2])
        if n >= 5 and head2 in COLON_FORMULAS:
            emit(words[1].end, ":", "T10", "Ikki nuqta",
                 "'%s' formulasidan keyin izohlash ikki nuqta bilan beriladi"
                 % " ".join(t.text for t in words[:2]), 0.72)

        # --- T9: "Assalomu alaykum, aziz do'stim!" / "Salom, aziz
        # o'quvchilar!" - salomlashish + murojaat + undov belgisi
        if n > 4 and words[0].lower in ("assalomu", "salom"):
            if (words[0].lower == "assalomu" and n > 1
                    and words[1].lower == "alaykum"):
                k = self._voc_chain_end(words, 2)
            elif words[0].lower == "salom":
                k = self._voc_chain_end(words, 1)
            else:
                k = None
            if k is not None and n - (k + 1) >= 3:
                emit(words[k].end, "!", "T9", "Undov belgisi",
                     "Salomlashish va murojaatdan keyin undov belgisi "
                     "qo'yildi", 0.72)
                cap_next(k + 1)

        # --- v7.2 (S8): ISM + 'siz/sen' - murojaatli yangi gap:
        # '... optimallashtirish. Azizbek, siz ... ko'rdingizmi?'
        for k in range(1, n - 2):
            w0 = words[k].lower
            if w0 not in KNOWN_UZ_NAMES:
                continue
            if words[k + 1].lower not in ("siz", "sen", "sizlar"):
                continue
            if (words[k + 2].lower in POSTPOSITIONS
                    or words[k + 2].lower in ("uchun", "bilan", "kabi",
                                              "ning", "ga", "da")):
                continue
            pv = words[k - 1].lower
            if not (is_finite(pv) or pv in PRED_NOMINAL or pv in COPULA_END
                    or masdar_like(pv) or has_cond_suffix(pv)
                    or self._is_imperative_word(pv)):
                continue
            if pv.endswith("ning"):
                continue
            if not has_emit(words[k - 1].end):
                emit(words[k - 1].end, ".", "S8", "Gap chegarasi",
                     "Ism-murojaatli yangi gap boshlandi ('%s, %s ...')"
                     % (words[k].text, words[k + 1].text), 0.78)
                cap_next(k)
            break

        # --- gap boshidagi tanlov so'rog'i: "Bugunmi, ertami yoki indinmi?"
        skip_until = -1
        q0 = self._question_stretch(words, 0)
        if q0 is not None and q0[1] == "alt":
            k = q0[0]
            emit(words[k].end, "?", "S2", "So'roq belgisi",
                 "Tanlov so'rogi oxiriga so'roq belgisi qo'yildi", 0.8)
            self._alt_commas(words, 0, k, out, emit)
            skip_until = k

        clause_start = 0
        for i in range(n - 1):
            if i < skip_until:
                continue
            w = words[i]
            wl = w.lower
            # v7.4 (G1): gap boshi salomlashuv formulasi + murojaat + keyingi
            # mustaqil gap: 'salom men bugun keldim' -> 'Salom! Men keldim.'
            # 'assalomu alaykum ustoz men keldim' -> 'Assalomu alaykum,
            # ustoz! Men keldim.' / 'yaxshimisiz do'stim nima qilyapsan' ->
            # 'Yaxshimisiz, do'stim? Nima qilyapsan?'
            if (i == 0 and wl in GREETING_OPENERS and n >= 3
                    and not has_emit(w.end)):
                ce_g = 0
                for t2 in range(1, min(n, 5)):
                    w2 = words[t2].lower
                    if (w2 in GREETING_OPENERS or w2 in VOC_ADJ
                            or w2 in VOC_HEADS or w2 in MUROJAAT_STRONG
                            or w2 in MUROJAAT_KIN):
                        ce_g = t2
                    else:
                        break
                k0 = ce_g + 1
                # salom + oddiy ot: 'salom dunyo men bugun keldim' — ot
                # murojaat bo'lishi mumkin, agar keyin ega-olmosh kelsa
                voc_extra = (ce_g == 0 and k0 < n
                             and self._nounish(words[k0])
                             and words[k0].lower not in POSTPOSITIONS
                             and k0 + 1 < n
                             and words[k0 + 1].lower in SUBJECT_PRONOUNS)
                grp_end = k0 if voc_extra else ce_g
                after = grp_end + 1
                if (after < n
                        and self._right_clause_len(words, after, 12) >= 2):
                    # v7.4: ajratish FAQAT murojaatsiz holatda va keyingi gap
                    # ega-olmosh bilan boshlanganda ('salom men bugun keldim'):
                    # murojaatli formulalar bitta gapda qoladi (etalon:
                    # 'assalomu alaykum, aziz do'stim, bugun kelaman.')
                    starts_new = (ce_g == 0 and
                                  (voc_extra
                                   or words[after].lower in SUBJECT_PRONOUNS
                                   or words[after].lower in DEMONSTRATIVES))
                    if starts_new:
                        ch_g = ("?" if wl.endswith(("mi", "misiz", "misan",
                                                    "misizlar")) else "!")
                        if voc_extra:
                            e_c = self._comma_after(words[ce_g], 0.72, "V3",
                                                    "Vergul",
                                                    "Salomlashuvdan keyingi "
                                                    "murojaat vergul bilan "
                                                    "ajratildi")
                            if e_c:
                                out.append(e_c)
                        emit(words[grp_end].end, ch_g, "S1",
                             "Gap chegarasi",
                             "Salomlashuv formulasi mustaqil gap sifatida "
                             "ajratildi", 0.74)
                        cap_next(after)
                        clause_start = after
                        continue
            ce = self._clause_end(wl)
            # v7 (S5): predikativ sifat kuchaytiruvchi bilan tugagan gapni
            # ham tugallangan kesim deb tanymiz: 'birinchi taassurot juda
            # muhim.' - lekin oldingi so'z ot (ega) bo'lsin, 'bizga juda
            # muhim mijoz' kabi aniqlash emas
            pred_adj_end = (wl in PRED_ADJ and i >= 2
                            and words[i - 1].lower in INTENSIFIERS
                            and self._nounish(words[i - 2])
                            and not words[i - 2].lower.endswith(
                                ("ga", "dan", "ning", "ni")))
            if not ce and not pred_adj_end:
                # v7.2 (S10b): SANA + HAFTA KUNI formulasi - gap oxiri:
                # 'Bugun 15-noyabr, yakshanba. Dastlabki ...'
                # (sana himoyalangan zona - so'z tokenlarida YO'Q, shuning
                # uchun oldingi so'z va oraliqdagi raqamlar tekshiriladi)
                if wl in WEEKDAYS and i >= 1 and i + 1 < n:
                    gap = self._text[words[i - 1].end:w.start]
                    has_date = any(c.isdigit() for c in gap)
                    prev_time = (words[i - 1].lower in TIME_ADVERBS
                                 or any(c.isdigit()
                                        for c in words[i - 1].text))
                    if (has_date and prev_time
                            and words[i + 1].lower not in POSTPOSITIONS
                            and self._right_has_predicate(words, i + 1, 10)):
                        e19 = self._comma_before(w, 0.8, "V19", "Vergul",
                                                 "Sana bilan hafta kuni orasiga "
                                                 "vergul qo'yildi")
                        if e19:
                            out.append(e19)
                        if not has_emit(w.end):
                            emit(w.end, ".", "S10", "Gap chegarasi",
                                 "Sana-hafta kuni formulasidan keyin yangi "
                                 "gap boshlandi ('%s')"
                                 % words[i + 1].text, 0.76)
                            cap_next(i + 1)
                        clause_start = i + 1
                        continue
                # v7.2: ravishdosh kesim + ega-olmosh: 'qurilgan bo'lib,
                # u hozirgacha ...' - vergul (S6 fe'l-birikmasiga xalaqt bermaydi)
                if (has_converb_suffix(wl) and i + 1 < n
                        and words[i + 1].lower in SUBJECT_PRONOUNS
                        and self._right_has_predicate(words, i + 1, 12)
                        and not has_emit(w.end)):
                    emit(w.end, ",", "S3", "Vergul",
                         "Ravishdosh kesimli qismdan keyin ega-olmoshli "
                         "davom vergul bilan ajratildi ('%s, %s ...')"
                         % (w.text, words[i + 1].text), 0.73)
                    clause_start = i + 1
                    continue
                # v7.2 (I9b): predikativ sifat + aniqlovchi olmosh izoh:
                # 'Javob oddiy: har birimiz ...'
                if ((wl in PRED_ADJ or wl == "oddiy") and i >= 1
                        and self._nounish(words[i - 1])
                        and i + 1 < n
                        and words[i + 1].lower in ("har", "hamma",
                                                   "quyidagi", "bular",
                                                   "shular")
                        and not has_emit(w.end)):
                    emit(w.end, ":", "I9", "Ikki nuqta",
                         "'%s' kesimidan keyin izoh ikki nuqta bilan "
                         "berildi" % w.text, 0.74)
                    clause_start = i + 1
                    continue
                # v7.2: qiyosiy '-roq' kesim ('undan ham muhimroq') + o'ngda
                # mustaqil kesim: '... muhimroq. Chiroyli dizayn kerak...'
                # (gohidan 'tezroq tugatish' kabi aniqlovchi holatda emas)
                # v7.5: chap qism kesimsiz bo'lsa ('men odatdagidan
                # ertaroq uygondim') bu qiyosiy ANIQLUVCHI, kesim emas
                if (wl.endswith("roq") and len(wl) >= 6 and i >= 1
                        and (words[i - 1].lower in ("ham", "hamda")
                             or words[i - 1].lower.endswith("dan"))
                        and i + 1 < n
                        and words[i + 1].lower not in POSTPOSITIONS
                        and self._right_has_predicate(words, i + 1, 12)
                        and left_complete(clause_start, i)
                        and not has_emit(w.end)):
                    emit(w.end, ".", "S5", "Gap chegarasi",
                         "Qiyosiy kesimli gap tugadi, yangi gap "
                         "boshlandi ('%s')" % words[i + 1].text, 0.74)
                    cap_next(i + 1)
                    clause_start = i + 1
                    continue
                # v7.2: DASH-mavzu kesimi: 'yana bir muhim masala xavfsizlik.
                # Foydalanuvchilarning ...' - mavzu otidan keyin yangi gap
                # (T8 qoidasi keyin tireni qo'yadi: 'masala - xavfsizlik.')
                if (i >= 2 and words[i - 1].lower in DASH_HEADS
                        and self._nounish(w)
                        and i + 1 < n
                        and words[i + 1].lower.endswith("ning")
                        and (words[i - 2].lower == "bir"
                             or self._adjlike(words[i - 2])
                             or words[i - 2].lower in PRED_ADJ)
                        and self._right_has_predicate(words, i + 1, 18)
                        and not has_emit(w.end)):
                    emit(w.end, ".", "S5", "Gap chegarasi",
                         "Mavzu-ot kesimidan keyin yangi gap boshlandi "
                         "('%s ... %s')" % (words[i - 1].text, w.text), 0.74)
                    cap_next(i + 1)
                    clause_start = i + 1
                    continue
                # v7.3 (I5b): tartib sanashning 2- va undan keyingi bo'laklari:
                # 'birinchisi, tahlil; ikkinchisi, ishlab chiqish; ...' -
                # ordinal oldidan nuqta-vergul (oldingi bo'lak ot bo'lak)
                if (wl in ENUM_POSS and i >= 2
                        and any(x.lower in ENUM_POSS for x in words[:i - 1])
                        and self._nounish(words[i - 1])
                        and not self._clause_end(words[i - 1].lower)
                        and not has_emit(words[i - 1].end)
                        and self._cb(w) != ";"):
                    emit(words[i - 1].end, ";", "I5", "Nuqta-vergul",
                         "Tartib sanash bo'laklari orasiga nuqta-vergul "
                         "qo'yildi", 0.85)
                    clause_start = i
                    continue
                # v7.3 (I4b): 'uchta bosqich mavjud: birinchisi, ...' -
                # 'mavjud' kesim sifatida ishlanmaydi, ammo tartib sanash
                # oldidan ikki nuqta baribir qo'yiladi
                if (wl in ("mavjud",) and i >= 1 and i + 1 < n
                        and words[i - 1].lower in ENUM_COLON_HEADS
                        and words[i + 1].lower in ENUM_POSS
                        and not has_emit(w.end)):
                    emit(w.end, ":", "I4", "Ikki nuqta",
                         "Sanab ko'rsatishdan oldin ikki nuqta qo'yildi "
                         "('(birinchisi, ...)' tartib sanashi)", 0.85)
                    clause_start = i + 1
                    continue
                # v7.3 (S5e): predikativ sifat + o'ngda o'z ega-si bilan
                # mustaqil gap: '... server barqarorligi bilan bog'liq.
                # Uni hal qilish uchun ...'
                # v7.3-2: 'eng muhim savolga o'tamiz' — 'eng' kuchaytiruvchi
                # oldida sifat ATTRIBUTIVE, kesim emas; to'ldiruvchi-shaklli
                # ot ('savolga') EGA bo'la olmaydi
                if (wl in PRED_ADJ and i >= 1 and i + 1 < n
                        and words[i - 1].lower != "eng"
                        and (i - clause_start + 1) >= 4
                        and not has_emit(w.end)):
                    rj = i + 1
                    rjw = words[rj].lower
                    case_obj = rjw.endswith(("ga", "ni", "da", "dan", "ning"))
                    pron_ok = (rjw in SUBJECT_PRONOUNS
                               or rjw in DEMONSTRATIVES
                               or self._is_pron_form(rjw))
                    noun_ok = (not case_obj
                               and self._nounish(words[rj])
                               and not self._adjlike(words[rj]))
                    subj_ok = pron_ok or noun_ok
                    # v7.3-3: o'ng qism buyruq mayli bo'lsa ega tushib qoladi
                    # ('... mashq qilish muhim. Ovqatni vaqtida yeng.') —
                    # bu holatda ham chegara QO'YILADI
                    if not subj_ok:
                        for tt in range(rj, min(n, rj + 5)):
                            if self._is_imperative_word(words[tt].lower):
                                subj_ok = True
                                break
                            if self._clause_end(words[tt].lower):
                                break
                    if (subj_ok and rjw not in POSTPOSITIONS
                            and rjw not in NO_BOUNDARY_AFTER_PRED
                            and not rjw.endswith("ning")
                            and self._right_clause_len(words, rj, 10) >= 2):
                        emit(w.end, ".", "S5", "Gap chegarasi",
                             "Predikativ sifatli gapdan keyin yangi mustaqil "
                             "gap boshlandi ('%s')" % words[rj].text, 0.73)
                        cap_next(rj)
                        clause_start = rj
                        continue
                # v7 (S1b): ot/masdar/predikativ sifat/'ega' bilan tugagan qism +
                # gap boshi formulasi: 'yaratish. To'g'risini aytsam...',
                # 'qiyin. Aytgancha, ...', 'tarixga ega. Masalan, ...'
                # v7.5: chap qism haqiqiy kesim bilan tugashi KERAK (kesimsiz
                # 'men odatdagidan ertaroq' kabi qismlar tugallangan emas) va
                # o'ngda yopilmagan so'roq bo'lmasa ('nega dasturing ...')
                if (wl in PRED_ADJ or wl in PRED_NOMINAL or wl == "ega"
                        or masdar_like(wl) or self._nounish(w)):
                    conf2, _ph = self._starter_match(words, i + 1)
                    min_conf = 0.73 if (wl in PRED_ADJ or wl in PRED_NOMINAL
                                        or wl == "ega"
                                        or masdar_like(wl)) else 0.78
                    # v7.2: boshlang'ich qism juda qisqa bo'lsa ('Masalan,
                    # server. Butunlay...') soxta chegara - kamida 2 so'z kerak
                    if (conf2 >= min_conf and not has_emit(w.end)
                            and (i - clause_start + 1) >= 2
                            and left_complete(clause_start, i)
                            and not self._never_start_word(words[i + 1].lower)
                            and not unresolved_qword(clause_start, i + 1)):
                        emit(w.end, ".", "S1", "Gap chegarasi",
                             "Tugallangan qismdan keyin yangi gap boshlanishi "
                             "aniqlandi ('%s' - gap boshi belgisi)"
                             % words[i + 1].text, conf2)
                        cap_next(i + 1)
                        clause_start = i + 1
                continue
            j = i + 1
            gap_pos = w.end
            prev_len = i - clause_start + 1
            blocked = has_emit(gap_pos)
            prev_imperative = self._is_imperative_word(wl)

            # Postpozitsiya: oldingi so'z uning to'ldiruvchisi (ot) -
            # gap tugamagan: 'muhim loyihamiz haqida ... gaplashamiz'
            if words[j].lower in POSTPOSITIONS:
                continue

            # Yordamchi kesimlar oldingi so'zga bog'lanadi:
            # 'kerak emas', 'mumkin emas', 'tayyor edi', 'kerak bo'ladi',
            # 'yetarli bo'lmasa', 'kerak bo'lsa' - birlik
            # v7.5: apostrofsiz shakllar ham ('bolgan', 'bolsa', 'bolsak')
            if words[j].lower in SUBJ_AUX_CONT:
                continue

            # v7.4: qo'shma kesim: ot + passiv yordamchi ('taqdim etildi',
            # 'e'lon qilindi', 'tavsiya etiladi') — o'rtasiga chegara YO'Q
            if words[j].lower in COMPOUND_PASS_AUX:
                continue

            # v7 (S6): FE'L BIRIKMASI QO'RIQLANADI: 'ishlay boshladi',
            # 'qilib yubordi', 'olib keldi', 'ko'ra boshlaydi' - ravishdosh
            # + yordamchi fe'l oralasiga HECH QANDAY belgi qo'yilmaydi
            # (v6 dagi 'ishlay. Boshladi' xatosi tuzatildi). Shart fe'li
            # ('kelmasa') va 'oldindan' kabi ravishlar chiqarib tashlanadi.
            if (is_converb_like(wl) and not has_cond_suffix(wl)
                    and words[j].lower not in VCHAIN_BLOCKLIST
                    and is_vchain_aux(words[j].lower)):
                continue

            # v7 (S4): ko'p so'zli kirish birikmasi aynan shu so'zda tugadi:
            # '... lozim, shunday ekan, avval ...' - chegara emas, birlikdan
            # keyin vergul ('shunday ekan', 'haqiqatan ham' frazalari)
            intro_done = False
            for kk in (3, 2):
                if i - kk + 1 >= 0:
                    phrase = " ".join(x.lower for x in words[i - kk + 1:i + 1])
                    if phrase in KIRISH_MULTI and i + 1 < n:
                        if not blocked:
                            emit(w.end, ",", "S4", "Vergul",
                                 "Kirish birikmasi '%s' dan keyin vergul "
                                 "qo'yildi" % phrase, KIRISH_MULTI[phrase])
                        clause_start = i + 1
                        intro_done = True
                        break
            if intro_done:
                continue

            # Yopish/minnatdorchilik formulasi - yangi gap:
            # '... berishingiz mumkin. Rahmat e'tiboringiz uchun.'
            # v7.3: '... kutaman. Hurmat bilan, Aziz.' - imzo formulasi
            if (words[j].lower in ("rahmat", "xayr", "omad", "tashakkur")
                    or (words[j].lower == "hurmat" and j + 1 < n
                        and words[j + 1].lower == "bilan")):
                if not blocked:
                    emit(gap_pos, ".", "S1", "Gap chegarasi",
                         "Minnatdorchilik/yopish formulasi yangi gapni "
                         "boshladi ('%s')" % words[j].text, 0.72)
                    cap_next(j)
                clause_start = j
                continue

            # A0) oldingi so'z '-mi' so'rogi - o'z oxiriga '?'
            #     (kirish so'zidan ustun turadi: 'mumkinmi? Sizningcha...')
            #     v7.5: keyingi so'z yangi gap boshi BO'LMASA ('... kelamanmi
            #     deb soradi') — o'zlashtirma nutq, so'roq belgisi QO'YILMAYDI
            if (self._mi_stem(wl) is not None and not blocked
                    and not self._never_start_word(words[j].lower)
                    and not any(x in ENUM_CONJ
                                for x in [t.lower for t in words[j:j + 5]])
                    and not any(self._mi_stem(t.lower, loose=True) is not None
                                for t in words[j:j + 6])):
                emit(gap_pos, "?", "S2", "So'roq belgisi",
                     "'-mi' so'roq yuklamasi bilan tugagan gap oxiriga so'roq "
                     "belgisi qo'yildi ('%s')" % w.text, 0.84)
                cap_next(j)
                clause_start = j
                continue

            # v7.2 (I7): kashf/qayd fe'li + izoh ro'yxati - IKKI NUQTA:
            # 'ayrim muammolar aniqlandi: ba'zi yozuvlari takrorlangan, ...'
            if (wl in DISCOVERY_VERBS and not blocked
                    and words[j].lower in ("bazi", "ayrim", "yana", "bir",
                                           "jumladan", "sabab", "sababi")
                    and self._right_has_predicate(words, j, 14)):
                emit(w.end, ":", "I7", "Ikki nuqta",
                     "'%s' dan keyin izoh ro'yxati ikki nuqta bilan "
                     "berildi" % w.text, 0.8)
                clause_start = j
                continue

            # v7.2 (I8): manfiy kesim + olmoshli izoh gap - IKKI NUQTA:
            # 'ish tugamaydi: u haqiqatan ham ishlashini sinab ko'rishimiz kerak'
            if (wl.endswith(("maydi", "maymiz", "mayman", "maysiz"))
                    and not blocked
                    and words[j].lower in ("u", "bu", "unda", "bunda",
                                           "uning", "buning")
                    and self._right_clause_len(words, j, 14) >= 6):
                emit(w.end, ":", "I8", "Ikki nuqta",
                     "Izoh gap oldidan ikki nuqta qo'yildi ('%s ...')"
                     % w.text, 0.76)
                clause_start = j
                continue

            # v7.2: buyruq kesim + VAQT ravishi - yangi gap:
            # '... tushuntirib bering. Kecha o'tkazilgan sinovda ...'
            if (prev_imperative and words[j].lower in TIME_ADVERBS
                    and not blocked
                    and self._right_has_predicate(words, j, 10)):
                emit(gap_pos, ".", "S1", "Gap chegarasi",
                     "Buyruq kesimdan keyin vaqt ravishi bilan yangi gap "
                     "boshlandi ('%s')" % words[j].text, 0.74)
                cap_next(j)
                clause_start = j
                continue

            # v7.2 (S10): SANA + HAFTA KUNI (kesimli yo'lda ham tekshiriladi)
            # - asosiy shoxi S10b (yozuv bo'lmagan qismda).

            # v7 (I4): sanab ko'rsatishdan oldin ikki nuqta:
            # 'muhokama qilamiz: birinchisi, server barqarorligi; ...'
            # v7.1: shu gapda oldin boshqa tartib so'z uchragan bo'lsa
            # (birinchisi ... ikkinchisi) bu O'RTA bo'lak - nuqta-vergul.
            if words[j].lower in ENUM_POSS and not blocked:
                if any(x.lower in ENUM_POSS for x in words[:j]):
                    emit(w.end, ";", "I5", "Nuqta-vergul",
                         "Tartib sanash bo'laklari orasiga nuqta-vergul "
                         "qo'yildi", 0.85)
                else:
                    emit(w.end, ":", "I4", "Ikki nuqta",
                         "Sanab ko'rsatishdan oldin ikki nuqta qo'yildi "
                         "('(birinchisi, ...)' tartib sanashi)", 0.85)
                clause_start = j
                continue

            # v7 (I5): mavjudlik fe'li + izohlash: 'yana bir masala bor: ...'
            # v7.2: keyingi so'z ot bo'lmasligi ham mumkin ('bor: har bir
            # vazifa ...') - shunchaki o'ng tarafta izoh kesimi bo'lsin kifoya
            # v7.3: 'mavjud' ham mavjudlik kesimi ('uchta bosqich mavjud: ...')
            if (wl in ("bor", "mavjud") and i >= 1
                    and words[i - 1].lower in ENUM_COLON_HEADS
                    and j < n
                    and words[j].lower not in ("emas", "edi", "ekan",
                                               "ham", "esa")
                    and words[j].lower not in POSTPOSITIONS
                    and not is_finite(words[j].lower)
                    and not self._is_imperative_word(words[j].lower)
                    and self._right_has_predicate(words, j, 22)
                    and not blocked):
                emit(w.end, ":", "I5", "Ikki nuqta",
                     "'%s bor' dan keyin izohlash ikki nuqta bilan "
                     "beriladi" % words[i - 1].text, 0.8)
                clause_start = j
                continue

            # v7.2 (I9): ot-kesim (zarur/kerak/kifoya...) + ro'yxat/izoh -
            # IKKI NUQTA: 'sinab ko'rishimiz zarur: osh, somsa va ...'
            if ((wl in PRED_NOMINAL) and i >= 1
                    and self._nounish(words[i - 1])
                    and words[j].lower not in SENT_START_STRONG
                    and words[j].lower not in KIRISH_STRONG
                    and words[j].lower not in ("agar", "agarda", "ammo",
                                               "lekin", "biroq", "chunki",
                                               "va", "ammo", "emas", "edi",
                                               "ekan", "bo'lsa", "bo'ladi",
                                               "bo'lmasa", "bo'lsa-da", "ham")
                    and words[j].lower not in POSTPOSITIONS
                    and not blocked):
                nxt5 = [t.lower for t in words[j:j + 4]]
                enum_va = ("va" in nxt5 and self._nounish(words[j])
                           and j + 1 < n and self._nounish(words[j + 1]))
                det_next = words[j].lower in ("har", "hamma", "quyidagi",
                                              "bular", "shular")
                if enum_va or det_next:
                    emit(w.end, ":", "I9", "Ikki nuqta",
                         "'%s' kesimidan keyin izoh/ro'yxat ikki nuqta "
                         "bilan berildi" % w.text, 0.74)
                    clause_start = j
                    continue

            # A) gap boshi formulasi: "avvalo", "xullas", "bundan tashqari"...
            conf, phrase = self._starter_match(words, j)
            if conf and (conf >= 0.71 or prev_len >= 6):
                if not (prev_imperative and conf < 0.76):
                    # v7.2: '-gan' kesimli RO'YXAT bo'lagi + 'yana bir' -
                    # bu ro'yxat davomi ('... tuzilgan, yana bir nechta...')
                    # - nuqta emas, VERGUL
                    list_cont = (phrase == "yana bir" and wl.endswith("gan"))
                    ch = "," if list_cont else "."
                    if not blocked:
                        emit(gap_pos, ch, "S1",
                             "Gap chegarasi" if ch == "." else "Vergul",
                             "Tugallangan gap qismidan keyin yangi gap "
                             "boshlanishi aniqlandi ('%s' - gap boshi "
                             "belgisi)" % words[j].text, conf)
                    if ch == ".":
                        cap_next(j)
                    if phrase in COLON_FORMULAS and j + 1 < n:
                        # "Esingizda bo'lsin:" - ikki nuqta formulasi
                        emit(words[j + 1].end, ":", "T10", "Ikki nuqta",
                             "'%s' formulasidan keyin izohlash ikki nuqta "
                             "bilan beriladi" % phrase, 0.72)
                        clause_start = j + 2
                        continue
                    clause_start = j
                    continue

            # B) to'g'ridan-to'g'ri so'roq gap
            # v7.5: yangi bo'lak gap-boshi so'zi bilan boshlanmasa yoki chap
            # qismda yopilmagan so'roq bo'lsa ('kim bilandir gaplashib
            # ketayotgan edi') — so'roq chegarasi QO'YILMAYDI
            q = self._question_stretch(words, j)
            if (q is not None and not blocked
                    and not self._never_start_word(words[j].lower)
                    and not unresolved_qword(clause_start, j)):
                k, kind = q
                if not prev_imperative:
                    # buyruqdan keyingi yumshoq so'roq ('aytingchi,
                    # sizningcha...') - nuqta emas, vergul (V2 beradi)
                    emit(gap_pos, ".", "S2", "Gap chegarasi",
                         "Yangi mustaqil so'roq gap boshlanishi uchun nuqta "
                         "qo'yildi", 0.78)
                    cap_next(j)
                emit(words[k].end, "?", "S2", "So'roq belgisi",
                     "To'g'ridan-to'g'ri so'roq gap oxiriga so'roq belgisi "
                     "qo'yildi", 0.82)
                if k + 1 < n:
                    cap_next(k + 1)
                if kind == "alt":
                    self._alt_commas(words, j, k, out, emit)
                clause_start = k + 1
                skip_until = k
                continue

            # C) buyruq mayli - yangi gap
            if (self._is_imperative_word(words[j].lower)
                    or (j + 1 < n
                        and self._is_imperative_word(words[j + 1].lower))):
                if not blocked:
                    emit(gap_pos, ".", "S1", "Gap chegarasi",
                         "Buyruq mayli yangi gapni boshladi ('%s')"
                         % words[j].text, 0.75)
                    cap_next(j)
                # "Esda saqlang:" - buyruqli ikki nuqta formulasi
                cphrase = " ".join(t.lower for t in words[j:j + 2])
                if cphrase in COLON_FORMULAS and j + 1 < n:
                    emit(words[j + 1].end, ":", "T10", "Ikki nuqta",
                         "'%s' formulasidan keyin izohlash ikki nuqta bilan "
                         "beriladi" % cphrase, 0.72)
                    clause_start = j + 2
                    continue
                clause_start = j
                continue
            # v7 (S5a): sifat-ravishdosh kesim + 'agar':
            # 'menejer bilan kelishilgan. Agar savollar chiqsa, ...'
            # v7.5: o'ngda 'degan/deb + kesim' nutq-ramkasi bo'lsa bu bitta
            # gap: '... ketaman agar vaqtim bolsa ... uchrashaman degan
            # matnni yubordi' — chegara QO'YILMAYDI
            if (words[j].lower in ("agar", "agarda")
                    and wl.endswith(("gan", "gen"))
                    and self._right_has_predicate(words, j)
                    and not blocked and not degan_frame_ahead(j)):
                emit(gap_pos, ".", "S5", "Gap chegarasi",
                     "Sifat-ravishdosh kesimli gap tugadi, shart ergash "
                     "gapli yangi gap boshlandi ('%s')" % words[j].text, 0.74)
                cap_next(j)
                clause_start = j
                continue

            # v7 (S5b): ot-kesim (kerak/lozim/mumkin/zarur/shart) yoki kopula
            # (edi/ekan/emas) bilan TUGALLANGAN gapdan keyin yangi mustaqil
            # gap: 'shu yerda ekan. Bir nechta...', 'kerak. Shuni ham ...'
            if (wl in COPULA_END or wl in PRED_NOMINAL or pred_adj_end):
                lw2 = words[j].lower
                # v7.2: 'yana bir ...' kirish birikmasi NO_BOUNDARY da emas:
                # '... kelishimiz mumkin. Yana bir muhim masala - ...'
                yana_bir = (lw2 == "yana" and j + 1 < n
                            and words[j + 1].lower == "bir")
                # v7.5: MODAL + bo'l- QO'SMA KESIM: 'kerak bo'lgan ishlarni',
                # 'kerak bo'ladi', 'yaratmoqchi bo'lsak' — kesim davomi,
                # chegara QO'YILMAYDI (apostrofsiz: 'bolgan', 'bolsak')
                if ((wl in PRED_NOMINAL or wl.endswith("moqchi"))
                        and lw2.startswith("bol")):
                    continue
                # v7.5: 'vaqt VA shart munosabatlari' — sanash bo'lagi,
                # 'shart' bu yerda ot, kesim emas
                if (wl in PRED_NOMINAL and i >= 2
                        and words[i - 1].lower in ENUM_CONJ):
                    continue
                # v7.3: 'shart bajarilmasa, ...' - ot-kesim + shart fe'li
                # bitta birlik ('shart' bu yerda EGA), chegara QO'YILMAYDI
                cond_after = (has_cond_suffix(lw2)
                              and lw2 not in ("bolsa", "bolmasa",
                                              "bolsa-da"))
                # v7.5: keyingi so'z yangi gap boshi BO'LMASA ('aytgan edi
                # degan jumlada', 'muhim ekanini tushundim') — davom
                if ((lw2 not in NO_BOUNDARY_AFTER_PRED or yana_bir)
                        and lw2 not in POSTPOSITIONS
                        and not cond_after
                        and not self._never_start_word(lw2)
                        and not unresolved_qword(clause_start, j)
                        and self._right_has_predicate(words, j)
                        and not blocked):
                    emit(gap_pos, ".", "S5", "Gap chegarasi",
                         "Tugallangan ot-kesimli gapdan keyin yangi mustaqil "
                         "gap boshlandi ('%s')" % words[j].text, 0.76)
                    cap_next(j)
                    clause_start = j
                    continue

            # v7 (S5c): tugallangan gap + ega-olmosh yangi gapda:
            # 'keldi. Men log fayllarini ...' / 'kerak. Ular soat o'nda ...'
            if words[j].lower in SUBJECT_PRONOUNS:
                nxt2 = words[j + 1].lower if j + 1 < n else ""
                rcl = self._right_clause_len(words, j)
                # v7.3 (S9-DA): '"Tayyor", dedi u, hujjatlari ham
                # rasmiylashtirildi.' - nutq-keltiruvchi fe'l + olmosh:
                # muallif gapi vergul bilan ajratiladi (nuqta EMAS)
                # 1-vergul NUTQ FE'LIDAN OLDIN, 2-vergul olmoshdan keyin
                if (wl in SPEECH_VERBS and 2 <= rcl <= 4
                        and nxt2 not in ENUM_CONJ
                        and nxt2 not in ("ham", "esa", "gina")
                        and j + 2 < n and not blocked
                        and i >= 1
                        and self._text[words[i - 1].end:words[i].start].strip() == ""):
                    emit(words[i - 1].end, ",", "S9", "Vergul",
                         "Muallif gapi ('%s %s ...') vergul bilan "
                         "ajratildi" % (words[i].text, words[j].text), 0.78)
                    emit(words[j].end, ",", "S9", "Vergul",
                         "Muallif gapidan ('%s') keyin vergul qo'yildi"
                         % words[j].text, 0.78)
                    clause_start = j + 1
                    continue
                if (nxt2 not in ("ham", "esa", "gina")
                        and nxt2 not in POSTPOSITIONS
                        and self._right_clause_len(words, j) >= 3
                        and not blocked
                        and not unresolved_qword(clause_start, j)):
                    emit(gap_pos, ".", "S5", "Gap chegarasi",
                         "Tugallangan gapdan keyin ega-olmosh bilan yangi gap "
                         "boshlandi ('%s')" % words[j].text, 0.76)
                    cap_next(j)
                    clause_start = j
                    continue

            # v7.3 (S5d): kesim + 'agar' shart ergash gap:
            # agar-ergash gap O'NGDA o'z asosiy qismi bilan kelganda bu
            # YANGI GAP: '...kechiktiramiz. Agar tarmoq uzilsa, zaxira
            # serverga o'tamiz.' Oxirida kelsa ('..., agar kerak bo'lsa')
            # - vergul (V8), chegara yo'q.
            # v7.5: o'ngda 'degan/deb + kesim' nutq-ramkasi bo'lsa bitta gap
            if (words[j].lower in ("agar", "agarda")
                    and prev_len >= 2 and not blocked
                    and not degan_frame_ahead(j)
                    and (is_finite(wl) or wl in PRED_NOMINAL
                         or wl in COPULA_END or pred_adj_end
                         or self._is_imperative_word(wl))):
                if self._agar_leads(words, j):
                    emit(gap_pos, ".", "S5", "Gap chegarasi",
                         "Shart ergash gapli yangi mustaqil gap boshlandi "
                         "('%s ...')" % words[j].text, 0.76)
                    cap_next(j)
                    clause_start = j
                    continue

            # v7.3 (S7): fe'l-kesim (yoki buyruq mayli) + o'z ega-si bilan
            # yangi mustaqil gap: '... tezligi ikki barobar oshdi.
            # Foydalanuvchilar shikoyatlari to'xtadi.'
            # (o'ng qismida mustaqil kesim bo'lishi shart)
            # v7.5: qattiq himoyalar — quyida batafsil
            if ((is_finite(wl) or self._is_imperative_word(wl))
                    and not wl.endswith("gan")
                    and wl not in SPEECH_VERBS
                    and prev_len >= 3 and not blocked):
                wj = words[j].lower
                # v7.5: yangi gap boshi shu so'z BO'Lolmaydi ('bolsak uni ...',
                # 'gaplashib ...', 'ekanini ...', 'va ...')
                if not self._never_start_word(wj):
                    # v7.5: yopilmagan so'roq ('nega dasturing hali ham xato
                    # qilmoqda deb soradi') — o'zlashtirma nutq, chegara YO'Q
                    if not unresolved_qword(clause_start, j):
                        # v7.5: 'ishlayman so'zi qatnashgan' — matn ichidagi
                        # so'zga ishora, yangi ega EMAS
                        meta_word = (wj in ("sozi", "sozlari")
                                     and j + 1 < n
                                     and (is_finite(words[j + 1].lower)
                                          or words[j + 1].lower.endswith("gan")))
                        # v7.5: genetiv-egalik ramki: 'Mening asosiy
                        # maqsadim ...' — 'maqsadim' egalik-ot, kesim EMAS
                        poss_frame = (
                            words[clause_start].lower in (
                                "mening", "bizning", "sening", "sizning",
                                "uning", "ularning")
                            and wl.endswith(("im", "imiz", "ing", "ingiz")))
                        subj_next = (self._nounish(words[j])
                                     or (self._adjlike(words[j])
                                         and words[j].lower not in INTENSIFIERS
                                         and j + 1 < n
                                         and self._nounish(words[j + 1])))
                        # v7.3: o'ng qism ot-kesim so'zidan boshlansa ('shart
                        # bajarilmasa, ...') - bu EGA bo'lagi, kesim undan keyin
                        rlen_from = j
                        if wj in PRED_NOMINAL and j + 1 < n:
                            rlen_from = j + 1
                        if (wj not in FUNCTION_WORDS
                                and wj not in NO_BOUNDARY_AFTER_PRED
                                and not wj.endswith("ning")
                                and not meta_word and not poss_frame
                                and subj_next
                                and self._right_clause_len(words, rlen_from, 12) >= 2):
                            emit(gap_pos, ".", "S7", "Gap chegarasi",
                                 "Fe'l-kesimli gapdan keyin o'z ega-si bilan yangi "
                                 "mustaqil gap boshlandi ('%s')" % words[j].text, 0.73)
                            cap_next(j)
                            clause_start = j
                            continue

            # D) bog'lovchi - chegara yo'q (V-qoidalar vergul qo'yadi)
            lw = words[j].lower
            if (lw in COMMA_TAKERS or lw in ENUM_CONJ
                    or lw in ("agar", "agarda", "zero",
                              "sababi", "ki", "deb")):
                clause_start = j
                continue

            # E) bog'lovchisiz tutash mustaqil gap
            # v7.5: keyingi so'z yangi gap boshi bo'lsa-DI ('ham', 'esa',
            # 'va', 'gaplashib', 'ekanini', 'bolsak') — chegara QO'YILMAYDI
            res = (None if self._never_start_word(words[j].lower)
                   else self._asyndetic(words, j, prev_len))
            if res is not None:
                # v7.5: egalik-ramki ('Mening asosiy maqsadim ... yaratish')
                # yoki matn-ichidagi so'zga ishora ('ishlayman so'zi ...') —
                # bu hali shu gapning davomi, vergul/nuqta QO'YILMAYDI
                if (words[clause_start].lower in ("mening", "bizning",
                                                  "sening", "sizning",
                                                  "uning", "ularning")
                        and wl.endswith(("im", "imiz", "ing", "ingiz"))):
                    clause_start = j
                    continue
                if (words[j].lower in ("sozi", "sozlari")
                        and is_finite(wl)):
                    clause_start = j
                    continue
                ch, conf, why = res
                if not blocked:
                    # v7.5: 'bu/shu/bular' bilan davom — ko'pincha yangi gap
                    # ('... dam oldim. Bu jumlalarda ...')
                    if ch == "," and words[j].lower in DEMONSTRATIVES:
                        ch = "."
                    # v7.5: chap qism 'agar/garchi' bilan boshlangan va uzun
                    # bo'lsa — asosiy qism tugadi, yangi gap boshlanadi
                    if (ch == "," and prev_len >= 4
                            and any(words[t].lower in ("agar", "agarda",
                                                       "garchi", "garchand")
                                    for t in range(clause_start, j))):
                        ch = "."
                    emit(gap_pos, ch, "S3",
                         "Gap chegarasi" if ch == "." else "Vergul", why, conf)
                    if ch == ".":
                        cap_next(j)
                clause_start = j
        return out

        enum_idx = [i for i, w in enumerate(words) if w.lower in ENUM_POSS]
        if len(enum_idx) < 2:
            enum_idx = []      # nuqta-vergul qismi o'tkazib yuboriladi,
                               # lekin pastdagi I5b ('yana bir eslatma:')
            # shoxi baribir ishlaydi

                continue     # allaqachon nuqta-vergul/chegara bor
            if "," in gap:

                m = re.compile(r",\s*").search(gap)

                                    prev.end + m.end(), "; ", "I5",
                                    "Nuqta-vergul",
                                    "Murakkab tartib sanash bo'laklari orasiga "
                                    "nuqta-vergul qo'yildi", 0.8))

                                "nuqta-vergul qo'yildi", 0.8))

            if (t.lower in ENUM_COLON_HEADS
                    and words[i - 1].lower in ("bir", "yana", "muhim")
                    and words[i + 1].lower not in POSTPOSITIONS
                    and not is_finite(words[i + 1].lower)
                    and words[i + 1].lower != "bor"
                    and any(self._is_imperative_word(x.lower)
                            or x.lower in PRED_NOMINAL
                            for x in words[i + 1:i + 4])
                    and not self._colon_in_gap(t)):
                out.append(Edit(t.end, t.end, ":", "I5", "Ikki nuqta",
                                "'%s' so'zidan keyin izohlash ikki nuqta "
                                "bilan beriladi" % t.text, 0.72))

    MASDAR_TAILS = ("shimiz", "shingiz", "shingizlar", "shilari", "shimi",

    def _tail_class(self, w: str) -> Optional[str]:
        """Masdar-egalik/osma-kesim dumi klassi ('shimiz', 'ligi'...)."""
        for t in self.MASDAR_TAILS:
            if w.endswith(t) and len(w) - len(t) >= 4:
                return t
        return None

                gapw = [x.lower for x in words[m + 1:a]]

                       for x in gapw):

                    out.append(e)

                break
            if found:
                break

    def _r_T8(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:

        'Yana bir muhim masala - xavfsizlik hisoblanadi.' /
        v7.2: kesimsiz shakl ham: 'Yana bir muhim masala - xavfsizlik.'
        v7.5: mavjudlik/modal kesimlar ('masala bor', 'masala kerak') —

        words = self._words(toks)
        for i in range(1, len(words) - 1):
            if words[i].lower not in DASH_HEADS:
                continue
            nxt = words[i + 1]
            if not self._nounish(nxt):
                continue
            if nxt.lower in DASH_HEADS:
                continue     # 'muhim masala.' - tire emas
            # v7.5: 'masala bor/kerak/mavjud...' — kesim, ta'rif emas
            if nxt.lower in ("bor", "mavjud", "kerak", "lozim", "zarur",
                             "shart", "mumkin", "kifoya", "yetarli",
                             "darkor", "iborat"):
                continue
            if self._adjlike(nxt) and not nxt.lower.endswith("lik"):
                continue     # 'qiyin' kabi sifat - tire emas
            # kesim aynan bosh so'zdan 2 so'z keyin bo'lsin:
            # 'masala - xavfsizlik hisoblanadi' OK,
            # 'savol javob bo'limi bo'ladi' (kesim uzoqda) - tire emas
            if i + 2 < len(words) and words[i + 2].lower not in COPULAS:
                continue
            gap = self._text[words[i].end:nxt.start]
            if gap.strip():
                continue
            out.append(Edit(words[i].end, words[i].end, " —", "T8",
                            "Tire",
                            "Ta'rif konstruksiyasida tire qo'yildi "
                            "('%s — %s')" % (words[i].text, nxt.text),
                            0.72))
            break

    def _v10_pairs(self, words: List[Tok], out: List[Edit]) -> None:
        """D (v7): (sifat+ot) juftliklari sanashi.
        'qizil olma, sariq nok va yashil uzum oldim' - uch va undan ortiq
        juftlik bo'lsa, juftliklar orasiga vergul qo'yiladi (yakka 'va'
        oldida vergul qo'yilmaydi)."""
        n = len(words)
        if n < 7:
            return
        pairs: List[Tuple[Tok, Tok]] = []
        i = 0
        while i < n - 1:
            if ((self._adjlike(words[i]) or masdar_like(words[i].lower))
                    and not is_finite(words[i].lower)
                    and self._nounish(words[i + 1])
                    and not self._adjlike(words[i + 1])
                    and not masdar_like(words[i + 1].lower)):
                pairs.append((words[i], words[i + 1]))
                i += 2
            else:
                i += 1
        if len(pairs) < 3:
            return
        # juftliklar orasida faqat bo'shliq/vergul tursin (oxirgi oraliqda
        # 'va/yoki' ham ruxsat - u quyida alohida tekshiriladi)
        for k in range(len(pairs) - 2):
            a_noun, b_adj = pairs[k][1], pairs[k + 1][0]
            gap = self._text[a_noun.end:b_adj.start]
            if not re.fullmatch(r"[ ,]*", gap):
                return
        # oxirgi juftlik oldida 'va/yoki/hamda' bo'lsin (yakka bog'lovchi)
        conj_ok = False
        for conj in ENUM_CONJ:
            # oxirgi juftlik oldidagi juftlik bilan orasida 'va' bo'lsin
            gap_last = self._text[pairs[-2][1].end:pairs[-1][0].start]
            if re.search(r"\b" + re.escape(conj) + r"\b", gap_last):
                conj_ok = True
                break
        if not conj_ok:
            return
        classes = {end_class(p[1].lower) for p in pairs}
        if len(classes) > 1 and "" not in classes:
            return
        # vergul faqat 'va'dan UZOQDA turgan juftliklardan keyin qo'yiladi:
        # 'qizil olma, sariq nok VA yashil uzum' - 'nok'dan keyin vergul YO'Q
        # (yakka 'va' oldida vergul qo'yilmaydi)
        for adj, noun in pairs[:-2]:
            if self._ca(noun) == "," or self._ca(noun) in ".!?…;:":
                continue
            e = self._comma_after(noun, 0.75, "V10", "Vergul",
                                  "Uyushiq (sifat+ot) juftliklari orasiga "
                                  "vergul qo'yildi")
            if e:
                out.append(e)

    def _prev_nonspace(self, tok: Tok) -> str:
        """Tokendan oldingi bo'sh bo'lmagan belgi (vergul/nuqta tekshiruvi)."""
        i = tok.start - 1
        while i >= 0 and self._text[i] in " \t":
            i -= 1
        return self._text[i] if i >= 0 else ""

    def _r_V19(self, s: Sentence, toks: List[Tok], out: List[Edit]) -> None:
        """V19 (v7): vaqt ravishi + hafta kuni appozitsiyasi.
        'Ertaga, dushanba kuni soat 10:00 da, yig'ilish bo'ladi.'

        n = len(words)
        if n < 5:

        for i in range(min(8, n - 4)):
            w = words[i]
            if (w.lower in TIME_ADVERBS and i + 1 < n
                    and words[i + 1].lower in WEEKDAYS
                    and (i == 0 or self._prev_nonspace(w) in ",")):
                base = i
                break
        if base is None:
            return
        w0, w1 = words[base], words[base + 1]
        if self._ca(w0) == ",":
            return
        endi = None
        for k in range(base + 2, min(n, base + 8)):
            w = words[k].lower
            if w == "da" or (w.endswith("da") and len(w) >= 4
                             and w not in POSTPOSITIONS):
                if w not in ("beri", "boshqa", "yana"):
                    endi = k
                    break
            if is_finite(w):
                break
        if endi is None or endi + 1 >= n:
            return
        after = words[endi + 1]
        if not (self._nounish(after) or is_finite(after.lower)):
            return
        if self._ca(words[endi]) == ",":
            return
        e1 = self._comma_after(w0, 0.75, "V19", "Vergul",
                               "Vaqt ravishi appozitsiyadan vergul bilan "
                               "ajratildi ('%s')" % w1.text)
        if e1:
            out.append(e1)
        if words[endi] is not words[-1]:
            e2 = self._comma_after(words[endi], 0.75, "V19", "Vergul",
                                   "Appozitsiya ('%s kuni ...') ikkinchi "
                                   "vergul bilan yopildi" % w1.text)
            if e2:
                out.append(e2)

    def _v10_adjectives(self, words: List[Tok], out: List[Edit]) -> None:
        """Ketma-ket uyushiq sifatlar: 'ishonchli, xavfsiz, tushunarli va ...'

        run: List[Tok] = []

        def flush(r: List[Tok], after: Optional[Tok] = None) -> None:
            # v7: 'va' dan keyin ham sifat kelsa va undan oldin >= 2 sifat
            # bo'lsa — bu sanash: 'qizil, ko'k va yashil'
            min_len = 3
            if (after is not None and after.lower in ENUM_CONJ and len(r) >= 2):
                ai = words.index(after) if after in words else -1
                if ai >= 0 and ai + 1 < len(words) and self._adjlike(words[ai + 1]):
                    min_len = 2
            if len(r) >= min_len:
                for a, b in zip(r, r[1:]):
                    if "," in self._text[a.end:b.start]:
                        continue
                    e = self._comma_before(b, 0.72, "V10", "Vergul",
                                           "Uyushiq sifatlar sanashi orasiga "
                                           "vergul qo'yildi")
                    if e:
                        out.append(e)

        for t in words:
            if (self._adjlike(t) and not is_finite(t.lower)
                    and t.lower not in INTENSIFIERS):
                # v7.2: '-lik/-gi' ot + keyingi ot = qoshma ot ('xavfsizlik
                # zaifligi') - sanash EMAS, run shu yerda uziladi
                if (t.lower.endswith(("lik", "gi", "g'i"))
                        and self._nounish(t)):
                    flush(run, t)
                    run = []
                    continue
                run.append(t)
            else:
                flush(run, t)
                run = []
        flush(run)

def _snippet(base: str, e: Edit) -> Tuple[str, str]:
    """O'zgarish atrofidagi matn parchasi: (eski, yangi)."""
    if not base:
        return "", ""
    i, j = e.start, e.end
    lo = max(0, i - 24)
    ro = min(len(base), j + 24)
    old = base[lo:i] + "«" + base[i:j] + "»" + base[j:ro]
    new = base[lo:i] + "«" + e.text + "»" + base[j:ro]
    return old, new


def human_report(r: Result, quiet: bool = False) -> str:
    if quiet:
        return r.corrected
    W = 64
    lines = []
    lines.append("=" * W)
    lines.append(" OQIM v%s — O'zbek tili punktuatsiya dvigateli" % __version__)
    st = r.stats or {}
    lines.append(" Rejim: %s | Chegara: %.2f | Gaplar: %s"
                 % (st.get("mode", "?"), st.get("threshold", 0.7),
                    st.get("sentences", "?")))
    lines.append("=" * W)
    lines.append("ASL MATN:")
    for ln in r.original.split("\n"):
        lines.append("  " + ln)
    lines.append("")
    lines.append("NATIJA:")
    for ln in r.corrected.split("\n"):
        lines.append("  " + ln)
    lines.append("")
    lines.append("O'ZGARISHLAR (%d):" % len(r.findings))
    for k, f in enumerate(r.findings, 1):
        old, new = _snippet(f.base, f)
        lines.append("  %d. [%s] %s (ishonch %.2f)" % (k, f.rule, f.name, f.conf))
        if f.msg:
            lines.append("     %s" % f.msg)
        if old:
            lines.append("     %s" % old)
            lines.append("     %s" % new)
    if not r.findings:
        lines.append("  (o'zgarish talab etilmadi)")
    lines.append("")
    lines.append("OGOHLANTIRISHLAR / TAVSIYALAR (%d):" % len(r.warnings))
    for k, f in enumerate(r.warnings, 1):
        old, new = _snippet(f.base, f)
        lines.append("  %d. [%s] %s (ishonch %.2f)" % (k, f.rule, f.name, f.conf))
        if f.msg:
            lines.append("     %s" % f.msg)
        if old:
            lines.append("     %s" % old)
    if not r.warnings:
        lines.append("  (shubhali joy topilmadi)")
    lines.append("")
    lines.append("STATISTIKA: %d ta o'zgarish, %d ta ogohlantirish, qoidalar: %s"
                 % (len(r.findings), len(r.warnings),
                    ", ".join(st.get("rules_applied", [])) or "-"))
    lines.append("=" * W)
    return "\n".join(lines)


def print_rules() -> None:
    print("=" * 70)
    print(" OQIM v%s — QOIDALAR KATALOGI (%d qoida)" % (__version__, len(RULEBOOK)))
    print("=" * 70)
    cur = ""
    for rb in RULEBOOK:
        if rb["cat"] != cur:
            cur = rb["cat"]
            print("")

        print("  [%s] %s" % (rb["id"], rb["name"]))
        print("      %s" % rb["desc"])
        print("      Misol: %s" % rb["ex"])
    print(QOLLANMA)


def run_demo(engine: UzbekGrammar) -> None:
    print("=" * 70)
    print(" OQIM v%s — DEMONSTRATSIYA (%d namuna)" % (__version__, len(DEMO)))
    print("=" * 70)
    for i, (text, note) in enumerate(DEMO, 1):
        r = engine.correct(text)
        print("")
        print("%2d) ASL:    %s" % (i, text))
        print("   NATIJA: %s" % r.corrected)
        rules = ", ".join(sorted({f.rule for f in r.findings})) or "-"
        warns = ", ".join(sorted({w.rule for w in r.warnings})) or ""
        print("   Qoidalar: %s%s" % (rules, (" | Ogohlantirish: " + warns) if warns else ""))
        print("   Izoh: %s" % note)


def run_tests() -> int:
    print("OQIM o'z-o'zini tekshirishi (v%s)..." % __version__)
    fails = 0
    eng = UzbekGrammar(mode="safe")
    for i, (src, expected) in enumerate(TESTS, 1):
        r = eng.correct(src)
        ok = r.corrected == expected
        if not ok:
            fails += 1
            print("FAIL %d:" % i)
            print("  kirdi:    %r" % src)
            print("  kutilgan: %r" % expected)
            print("  chiqdi:   %r" % r.corrected)
            for f in r.findings:
                print("    [%s] %s (%.2f)" % (f.rule, f.msg, f.conf))
        else:
            print("OK   %2d  %s" % (i, src))
    # Idempotentlik: natijani yana ishlovdan o'tkazsak — o'zgarmasligi kerak
    idem_fails = 0
    for src, _ in TESTS:
        once = eng.correct(src).corrected
        twice = eng.correct(once).corrected
        if once != twice:
            idem_fails += 1
            print("IDEMPOTENTLIK BUZILDI: %r -> %r -> %r" % (src, once, twice))
    # Aggressive rejim ham ishlashi kerak
    aggr = UzbekGrammar(mode="aggressive")
    for src, _ in TESTS:
        aggr.correct(src)
    print("")
    if fails or idem_fails:
        print("NATIJA: %d ta test muvaffaqiyatsiz, %d ta idempotentlik xatosi."
              % (fails, idem_fails))
        return 1
    print("NATIJA: BARCHA %d TEST MUVAFFAQIYATLI (+idempotentlik)." % len(TESTS))
    return 0

                   default="keep", help="apostrof uslubi (default: keep)")
    p.add_argument("--demo", action="store_true", help="barcha qoidalarni namoyish")
    p.add_argument("--bench", action="store_true",
                   help="katta matn (kitob hajmi) tezlik va barqarorlik testi")
    p.add_argument("--rules", action="store_true", help="qoidalar katalogi")
    p.add_argument("--test", action="store_true", help="o'z-o'zini tekshirish")
    p.add_argument("--version", "-v", action="version",
                   version="OQIM " + __version__)
    return p


def run_bench(engine: UzbekGrammar, repeats: int = 400) -> None:
    """Katta matn (kitob hajmi) testi: oltita etalon matnning xom
    (punksiyatsiyasiz) nusxalarini takrorlab, tezlik, idempotentlik va
    xotira samaradorligini o'lchaydi."""
    import time
    import gc

    base = [src_ for src_, _ in TESTS if len(src_.split()) >= 4]
    big = " ".join(base)
    text = " ".join([big] * repeats)
    n_kb = len(text.encode("utf-8")) / 1024.0
    print("=" * 64)
    print(" OQIM v%s -- KITOB HAJMI TESTI" % __version__)
    print(" Matn: %.1f KB, %d ta belgi" % (n_kb, len(text)))
    print("=" * 64)
    gc.collect()
    t0 = time.time()
    r1 = engine.correct(text)
    t1 = time.time()
    print(" 1-o'tish : %.2f s  (%d ta o'zgarish)" % (t1 - t0, len(r1.findings)))
    print("            tezlik: %.0f KB/s" % (n_kb / max(t1 - t0, 1e-9)))
    t0 = time.time()
    r2 = engine.correct(r1.corrected)
    t2 = time.time()
    print(" 2-o'tish : %.2f s  (tayyor matn - tezroq bo'lishi kerak)"
          % (t2 - t0))
    ok = r1.corrected == r2.corrected
    print(" Idempotentlik: %s" % ("OK" if ok else "BUZILGAN!"))
    if not ok:
        for i, (a, b) in enumerate(zip(r1.corrected, r2.corrected)):
            if a != b:
                print("  birinchi farq: %d: %r vs %r"
                      % (i, r1.corrected[i:i + 40], r2.corrected[i:i + 40]))
                break
    print("=" * 64)


def main(argv: Optional[List[str]] = None) -> int:
    args = build_parser().parse_args(argv)
    if args.rules:
        print_rules()
        return 0
    if args.test:
        return run_tests()
    engine = UzbekGrammar(mode="aggressive" if args.aggressive else "safe",
                          apostrophe=args.apostrophe)
    if args.bench:
        run_bench(engine)
        return 0
    if args.demo:
        run_demo(engine)
        return 0
    # Matn manbai
    if args.file:
        with open(args.file, "r", encoding="utf-8") as fh:
            text = fh.read()
    elif args.text:
        text = " ".join(args.text)
    elif not sys.stdin.isatty():
        text = sys.stdin.read()
    else:
        build_parser().print_help()
        return 2
    r = engine.correct(text)
    if args.json:
        output = json.dumps(r.as_dict(), ensure_ascii=False, indent=2)
    else:
        output = human_report(r, quiet=args.quiet)
    if args.out:
        with open(args.out, "w", encoding="utf-8") as fh:
            fh.write(output)
        if not args.quiet:
            print("Natija '%s' fayliga yozildi." % args.out)
    else:
        print(output, end="" if output.endswith("\n") else "\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
